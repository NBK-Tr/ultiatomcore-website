add_action( 'template_redirect', function () {
    if ( is_admin() ) return;
    $uri = isset( $_SERVER['REQUEST_URI'] ) ? wp_unslash( $_SERVER['REQUEST_URI'] ) : '';
    $path = trim( parse_url( $uri, PHP_URL_PATH ) ?: '', '/' );
    $path = strtolower( $path );
    if ( in_array( $path, [ 'klavye-savasi', 'kelime-savasi' ], true ) ) {
        wp_safe_redirect( home_url( '/klavye-muharebesi/' ), 301 );
        exit;
    }
} );

function uac_get_real_ip() {
    $headers = [ 'HTTP_CF_CONNECTING_IP', 'HTTP_X_FORWARDED_FOR', 'HTTP_X_REAL_IP', 'REMOTE_ADDR' ];
    foreach ( $headers as $h ) {
        if ( ! empty( $_SERVER[ $h ] ) ) {
            $ip = trim( explode( ',', sanitize_text_field( wp_unslash( $_SERVER[ $h ] ) ) )[0] );
            if ( filter_var( $ip, FILTER_VALIDATE_IP ) ) return $ip;
        }
    }
    return '0.0.0.0';
}

}

function uac_normalize_text( $text ) {
    $text = mb_strtolower( (string) $text, 'UTF-8' );
    $text = strtr( $text, [
        '4'=>'a','@'=>'a','â'=>'a','à'=>'a','á'=>'a','8'=>'b','©'=>'c','ç'=>'c',
        '0'=>'o','ö'=>'o','ø'=>'o','1'=>'i','!'=>'i','|'=>'i','ı'=>'i','î'=>'i',
        '3'=>'e','é'=>'e','ê'=>'e','7'=>'t','5'=>'s','$'=>'s','ş'=>'s','9'=>'g','ğ'=>'g',
        'ü'=>'u','û'=>'u','vv'=>'w','+'=>'t','><'=>'x','/-\\'=>'a','|<'=>'k',
        'а'=>'a','е'=>'e','о'=>'o','р'=>'p','с'=>'c','і'=>'i','ѕ'=>'s','ԁ'=>'d',
    ] );
    $text = preg_replace( '/(\w)\1+/u', '$1', $text );
    $text = str_replace( [ ' ', '-', '.', '_', '*', '/', '\\', '~', '^' ], '', $text );
    return $text;
}

    foreach ( $bad_words as $word ) {

    }

        if ( $clean === $word ) return $word;
    }
    return false;
}

function uac_ban_ip( $ip ) {
    $banned = get_option( 'uac_banned_ips', [] );
    if ( ! is_array( $banned ) ) $banned = [];
    if ( ! in_array( $ip, $banned, true ) ) {
        $banned[] = $ip;
        update_option( 'uac_banned_ips', array_values( $banned ), false );
    }
}

function uac_is_banned( $ip ) {
    $banned = get_option( 'uac_banned_ips', [] );
    return is_array( $banned ) && in_array( $ip, $banned, true );
}

function uac_log_violation( $ip, $type, $content, $triggered_word = '' ) {
    $log_key = 'uac_violations_' . md5( $ip );
    $violations = get_transient( $log_key );
    if ( ! is_array( $violations ) ) $violations = [];
    $violations[] = [
        'time'    => current_time( 'mysql' ),
        'type'    => sanitize_key( $type ),
        'content' => mb_substr( sanitize_text_field( $content ), 0, 160, 'UTF-8' ),
        'word'    => sanitize_text_field( $triggered_word ),
    ];
    $violations = array_slice( $violations, -80 );
    set_transient( $log_key, $violations, 7 * DAY_IN_SECONDS );

    $count_key = 'uac_violation_count_' . md5( $ip );
    $count = (int) get_transient( $count_key ) + 1;
    set_transient( $count_key, $count, 7 * DAY_IN_SECONDS );

    if ( $count >= 5 && in_array( $type, [ 'bad_word', 'cheat' ], true ) ) uac_ban_ip( $ip );
}

function uac_check_rate_limit( $ip, $action = 'save' ) {
    $configs = [
        'save'  => [ 'limit' => 10, 'window' => 60, 'ban_threshold' => 45 ],
        'chat'  => [ 'limit' => 8,  'window' => 60, 'ban_threshold' => 90 ],
        'score' => [ 'limit' => 10, 'window' => 60, 'ban_threshold' => 45 ],
        'check' => [ 'limit' => 25, 'window' => 60, 'ban_threshold' => 70 ],
        'start' => [ 'limit' => 18, 'window' => 60, 'ban_threshold' => 80 ],
        'daily' => [ 'limit' => 25, 'window' => 60, 'ban_threshold' => 90 ],
        'home'  => [ 'limit' => 60, 'window' => 60, 'ban_threshold' => 160 ],
    ];
    $cfg = $configs[ $action ] ?? $configs['save'];
    $key = 'uac_rate_' . sanitize_key( $action ) . '_' . md5( $ip );
    $data = get_transient( $key );
    if ( ! is_array( $data ) || time() - (int) $data['first'] > $cfg['window'] ) $data = [ 'count' => 0, 'first' => time() ];

    $data['count']++;
    set_transient( $key, $data, $cfg['window'] );

    if ( $data['count'] >= $cfg['ban_threshold'] ) {
        uac_ban_ip( $ip );
        uac_log_violation( $ip, 'spam', "Rate limit ban: {$action}" );
        return [ 'blocked' => true, 'retry_after' => $cfg['window'], 'banned' => true ];
    }
    if ( $data['count'] > $cfg['limit'] ) {
        $remaining = $cfg['window'] - ( time() - (int) $data['first'] );
        return [ 'blocked' => true, 'retry_after' => max( 1, $remaining ), 'banned' => false ];
    }
    return [ 'blocked' => false ];
}

function uac_validate_name( $name, $max = 30 ) {
    $name = trim( sanitize_text_field( $name ) );
    if ( mb_strlen( $name, 'UTF-8' ) < 2 || mb_strlen( $name, 'UTF-8' ) > $max ) return [ false, 'İsim 2-' . $max . ' karakter olmalıdır.' ];
    $norm_name = uac_normalize_text( $name );
    $reserved_names = [ 'admin', 'administrator', 'yonetici', 'yönetici', 'moderator', 'mod', 'root', 'system', 'sistem', 'support', 'destek' ];
    foreach ( $reserved_names as $reserved ) {
        if ( $norm_name === uac_normalize_text( $reserved ) ) return [ false, 'Bu isim sistem/yönetici adı olarak ayrılmıştır.' ];
    }
    $bad = uac_contains_bad_words( $name );
    if ( $bad ) return [ false, 'Uygunsuz kelime tespit edildi.' ];
    return [ true, $name ];
}

function uac_auth_player_key( $name ) {
    $key = sanitize_key( uac_normalize_text( $name ) );
    return $key !== '' ? $key : 'oyuncu';
}

function uac_player_auth_key( $name ) {

}

function uac_pin_from_request( WP_REST_Request $request ) {
    $pin = $request->get_param( 'player_pin' );
    if ( $pin === null || $pin === '' ) $pin = $request->get_param( 'pin' );
    return trim( sanitize_text_field( (string) $pin ) );
}

function uac_validate_player_pin( $pin ) {
    $pin = trim( sanitize_text_field( (string) $pin ) );
    if ( $pin === '' ) return [ false, 'Oyuncu PIN gerekli. Aynı adı başka cihazda kullanmak için bu PIN gerekir.' ];
    $len = mb_strlen( $pin, 'UTF-8' );
    if ( $len < 4 || $len > 12 ) return [ false, 'PIN 4-12 karakter olmalıdır.' ];
    if ( ! preg_match( '/^[a-zA-Z0-9_-]+$/u', $pin ) ) return [ false, 'PIN sadece harf, rakam, tire veya alt çizgi içerebilir.' ];
    return [ true, $pin ];
}

function uac_player_auth_get( $name ) {
    $data = get_option( uac_player_auth_key( $name ), null );
    return is_array( $data ) ? $data : null;
}

function uac_player_pin_verify_only( $name, $pin ) {
    [ $valid_pin, $pin_or_msg ] = uac_validate_player_pin( $pin );
    if ( ! $valid_pin ) return [ false, $pin_or_msg, 'pin_required' ];
    $data = uac_player_auth_get( $name );
    if ( ! is_array( $data ) || empty( $data['pin_hash'] ) ) return [ null, 'Bu oyuncu adı için henüz PIN belirlenmemiş.', 'pin_not_set' ];
    if ( ! wp_check_password( $pin_or_msg, (string) $data['pin_hash'] ) ) return [ false, 'Bu oyuncu adı için PIN yanlış.', 'pin_wrong' ];
    return [ true, '', 'pin_ok' ];
}

function uac_player_pin_claim_or_verify( $name, $pin, $source = 'game' ) {
    $name = sanitize_text_field( $name );
    [ $valid_pin, $pin_or_msg ] = uac_validate_player_pin( $pin );
    if ( ! $valid_pin ) return [ false, $pin_or_msg, 'pin_required' ];

    $key = uac_player_auth_key( $name );
    $data = get_option( $key, null );
    $now = current_time( 'mysql' );
    $source = sanitize_key( $source ?: 'game' );

    if ( is_array( $data ) && ! empty( $data['pin_hash'] ) ) {
        if ( ! wp_check_password( $pin_or_msg, (string) $data['pin_hash'] ) ) return [ false, 'Bu oyuncu adı için PIN yanlış. Bu adı başka biri kullanıyorsa farklı ad seç.', 'pin_wrong' ];
        $sources = isset( $data['sources'] ) && is_array( $data['sources'] ) ? $data['sources'] : [];
        $sources[ $source ] = $now;
        $data['name'] = $name;
        $data['sources'] = $sources;
        $data['updated'] = $now;
        update_option( $key, $data, false );
        return [ true, 'PIN doğrulandı.', 'pin_ok' ];
    }

    $data = [
        'name'      => $name,
        'pin_hash'  => wp_hash_password( $pin_or_msg ),
        'created'   => $now,
        'updated'   => $now,
        'sources'   => [ $source => $now ],
    ];
    update_option( $key, $data, false );
    return [ true, 'Oyuncu adı PIN ile kilitlendi.', 'pin_created' ];
}

function uac_request_claim_or_verify_pin( WP_REST_Request $request, $name, $source = 'game' ) {
    return uac_player_pin_claim_or_verify( $name, uac_pin_from_request( $request ), $source );
}

function uac_is_prime( $n ) {
    $n = (int) $n;
    if ( $n <= 1 ) return false;
    if ( $n <= 3 ) return true;
    for ( $i = 2; $i <= sqrt( $n ); $i++ ) if ( $n % $i === 0 ) return false;
    return true;
}

function uac_validate_password_solution( $solution ) {
    $errors = [];
    foreach ( uac_password_rules() as $index => $rule ) {
        [ $text, $check ] = $rule;
        try { $ok = (bool) $check( $solution ); }
        catch ( Throwable $e ) { $ok = false; }
        if ( ! $ok ) $errors[] = [ 'index' => $index, 'text' => $text ];
    }
    return [ empty( $errors ), $errors ];
}

function uac_pg_save_solution( WP_REST_Request $request ) {
    $ip = uac_get_real_ip();
    if ( uac_is_banned( $ip ) ) return uac_json_error( 'Erişiminiz engellenmiştir.', 'banned', 403 );

    $rate = uac_check_rate_limit( $ip, 'save' );
    if ( $rate['blocked'] ) return uac_json_error( $rate['banned'] ? 'Çok fazla istek nedeniyle erişiminiz engellendi.' : 'Çok fazla istek gönderdiniz.', 'rate_limited', 429, [ 'retry_after' => $rate['retry_after'] ] );

    $name = sanitize_text_field( $request->get_param( 'name' ) );
    $solution = (string) $request->get_param( 'solution' );
    if ( $solution === '' ) $solution = (string) $request->get_param( 'password' );
    $solution = sanitize_text_field( $solution );

    [ $valid_name, $name_or_msg ] = uac_validate_name( $name, 30 );
    if ( ! $valid_name ) return uac_json_error( $name_or_msg, 'invalid_name', 400 );
    $name = $name_or_msg;

    [ $pin_ok, $pin_msg, $pin_code ] = uac_request_claim_or_verify_pin( $request, $name, 'password_challenge' );
    if ( ! $pin_ok ) return uac_json_error( $pin_msg, $pin_code, 403 );

    if ( uac_strlen( $solution ) < 8 || uac_strlen( $solution ) > 500 ) return uac_json_error( 'Çözüm 8-500 karakter arasında olmalıdır.', 'invalid_solution', 400 );

    $bad_solution = uac_contains_bad_words( $solution );
    if ( $bad_solution ) {
        uac_log_violation( $ip, 'bad_word', "$name | $solution", $bad_solution );
        return uac_json_error( 'Uygunsuz kelime kullanımı tespit edildi.', 'bad_word', 400 );
    }

    [ $ok, $errors ] = uac_validate_password_solution( $solution );
    if ( ! $ok ) return uac_json_error( 'Çözüm tüm kuralları sağlamıyor.', 'rules_failed', 400, [ 'rule_errors' => $errors ] );

    $existing = get_posts( [
        'post_type'   => 'pg_solution',
        'post_status' => 'publish',
        'numberposts' => 1,
        'meta_query'  => [
            'relation' => 'AND',
            [ 'key' => '_uac_kind', 'value' => 'password_solution' ],
            [ 'key' => 'player_name', 'value' => $name ],
        ],
    ] );

    $post_id = $existing ? $existing[0]->ID : wp_insert_post( [
        'post_type'   => 'pg_solution',
        'post_status' => 'publish',
        'post_title'  => 'password_solution',
    ] );

    if ( is_wp_error( $post_id ) || ! $post_id ) return uac_json_error( 'Kayıt oluşturulamadı.', 'save_failed', 500 );

    update_post_meta( $post_id, '_uac_kind', 'password_solution' );
    update_post_meta( $post_id, 'player_name', $name );
    update_post_meta( $post_id, 'solution_text', $solution );
    update_post_meta( $post_id, 'solution_length', uac_strlen( $solution ) );
    update_post_meta( $post_id, 'rule_version', '4.0.0' );
    update_post_meta( $post_id, 'ip_hash', hash( 'sha256', $ip . wp_salt() ) );
    uac_chat_unlock_player( $name, 'password_challenge' );

    return rest_ensure_response( [ 'ok' => true, 'message' => 'Kaydedildi.' ] );
}

function uac_chat_clean_message( $message ) {
    $message = trim( preg_replace( '/\s+/u', ' ', sanitize_textarea_field( (string) $message ) ) );
    
    $message = preg_replace( '/\s+\[[a-z0-9]{2,5}\]$/iu', '', $message );
    return trim( $message );
}

function uac_chat_compact_text( $text ) {
    $text = mb_strtolower( (string) $text, 'UTF-8' );
    $text = strtr( $text, [
        'ı' => 'i', 'ğ' => 'g', 'ü' => 'u', 'ş' => 's', 'ö' => 'o', 'ç' => 'c',
        'İ' => 'i', 'Ğ' => 'g', 'Ü' => 'u', 'Ş' => 's', 'Ö' => 'o', 'Ç' => 'c',
    ] );
    $text = preg_replace( '/[^a-z0-9]/u', '', $text );
    return $text ?: '';
}

function uac_chat_short_allowlist() {
    return [
        'as','sa','slm','mrb','mrh','ii','iyi','ok','oki','tm','tmm','gg','wp','xd','lol','brb','afk','np','ty','thx','hi','hey','yo','no','yes','gl','hf','ez','wb','gm','gn','bb','by','js','php','css','html','sql','api','fps','lag','bug','fix','mod','bot','ai','pc','dc','yt','ig','hmm','hm','tsk','aq'
    ];
}

function uac_chat_is_low_quality( $text ) {
    $compact = uac_chat_compact_text( $text );
    if ( $compact === '' ) return [ false, '' ];
    $allowed = uac_chat_short_allowlist();
    if ( in_array( $compact, $allowed, true ) ) return [ false, '' ];

    $len = mb_strlen( $compact, 'UTF-8' );
    if ( $len === 1 ) return [ true, 'Tek harflik mesajlar spam gibi görünüyor.' ];
    if ( $len <= 2 ) return [ true, 'Mesaj çok kısa ve anlamsız görünüyor.' ];
    if ( preg_match( '/^(.)\1{2,}$/u', $compact ) ) return [ true, 'Aynı karakter tekrarı spam gibi görünüyor.' ];
    if ( preg_match( '/^(.{1,2})\1{2,}$/u', $compact ) ) return [ true, 'Tekrarlı harf dizisi spam gibi görünüyor.' ];
    if ( ! preg_match( '/[aeiou]/u', $compact ) && $len <= 6 ) return [ true, 'Mesaj rastgele harf dizisi gibi görünüyor.' ];
    if ( $len >= 4 && preg_match( '/^(asdf|sdfg|dfgh|fghj|qwer|wert|zxcv|xcvb|cvbn|hjkl)/u', $compact ) ) return [ true, 'Klavye karışık harfleri spam gibi görünüyor.' ];

    return [ false, '' ];
}

function uac_chat_device_key( $raw ) {
    $raw = sanitize_text_field( (string) $raw );
    $raw = preg_replace( '/[^a-zA-Z0-9_-]/', '', $raw );
    if ( strlen( $raw ) < 8 || strlen( $raw ) > 90 ) return 'no_device_' . wp_generate_password( 8, false, false );
    return $raw;
}

function uac_chat_recent_wait( $key, $seconds ) {
    $last = (int) get_transient( $key );
    if ( $last > 0 ) {
        $wait = $seconds - ( time() - $last );
        if ( $wait > 0 ) return $wait;
    }
    return 0;
}

function uac_chat_touch_recent( $key, $seconds ) {
    set_transient( $key, time(), $seconds );
}

function uac_chat_counter_check( $key, $limit, $window ) {
    $data = get_transient( $key );
    if ( ! is_array( $data ) || time() - (int) ( $data['first'] ?? 0 ) > $window ) {
        $data = [ 'first' => time(), 'count' => 0 ];
    }
    $data['count'] = (int) $data['count'] + 1;
    set_transient( $key, $data, $window );
    if ( $data['count'] > $limit ) {
        $retry = $window - ( time() - (int) $data['first'] );
        return max( 1, $retry );
    }
    return 0;
}

function uac_chat_player_key( $name ) {
    $key = sanitize_key( uac_normalize_text( $name ) );
    return $key !== '' ? $key : 'oyuncu';
}

function uac_chat_unlock_key( $name ) {
    return 'uac_chat_unlocked_' . uac_chat_player_key( $name );
}

function uac_chat_unlock_player( $name, $game = 'game' ) {
    $name = sanitize_text_field( $name );
    if ( $name === '' ) return false;
    $key = uac_chat_unlock_key( $name );
    $data = get_option( $key, [] );
    if ( ! is_array( $data ) ) $data = [];
    $games = isset( $data['games'] ) && is_array( $data['games'] ) ? $data['games'] : [];
    $game = sanitize_key( $game ?: 'game' );
    $now = current_time( 'mysql' );
    $games[ $game ] = $now;
    $data = [
        'name'       => $name,
        'unlocked'   => true,
        'games'      => $games,
        'last_game'  => $game,
        'updated'    => $now,
    ];
    update_option( $key, $data, false );
    return true;
}

function uac_chat_completed_by_km( $name ) {
    if ( ! function_exists( 'uac_km_profile_key' ) ) return false;
    $profile = get_option( uac_km_profile_key( $name ), null );
    if ( is_array( $profile ) && ( (int) ( $profile['total_games'] ?? 0 ) > 0 || (int) ( $profile['best_score'] ?? 0 ) > 0 ) ) return true;

    $entry_key = sanitize_key( mb_strtolower( $name, 'UTF-8' ) );
    foreach ( [ 'uac_km_leaderboard_all', 'uac_km_leaderboard_week_' . gmdate( 'oW' ), 'uac_km_leaderboard_day_' . gmdate( 'Ymd' ) ] as $board_key ) {
        $board = get_option( $board_key, [] );
        if ( is_array( $board ) && isset( $board[ $entry_key ] ) ) return true;
    }
    return false;
}

function uac_chat_completed_by_daily( $name ) {
    if ( function_exists( 'uac_secret_v2_profile_key' ) ) {
        $profile = get_option( uac_secret_v2_profile_key( $name ), null );
        if ( is_array( $profile ) && ( (int) ( $profile['answered'] ?? 0 ) > 0 || (int) ( $profile['score'] ?? 0 ) > 0 ) ) return true;
        if ( function_exists( 'uac_secret_v2_find_profiles_by_name' ) ) {
            foreach ( uac_secret_v2_find_profiles_by_name( $name ) as $old_profile ) {
                if ( is_array( $old_profile ) && ( (int) ( $old_profile['answered'] ?? 0 ) > 0 || (int) ( $old_profile['score'] ?? 0 ) > 0 ) ) return true;
            }
        }
    }

    if ( function_exists( 'uac_daily_get_profile_by_name' ) ) {
        $daily_profile = uac_daily_get_profile_by_name( $name );
        if ( is_array( $daily_profile ) && ( (int) ( $daily_profile['answered'] ?? 0 ) > 0 || (int) ( $daily_profile['score'] ?? 0 ) > 0 ) ) return true;
    }

    $entry_key = sanitize_key( mb_strtolower( $name, 'UTF-8' ) );
    foreach ( [ 'uac_secret_v2_rank_all', 'uac_daily_rank_all' ] as $board_key ) {
        $board = get_option( $board_key, [] );
        if ( ! is_array( $board ) ) continue;
        foreach ( $board as $key => $entry ) {
            $stored_key = sanitize_key( (string) $key );
            $stored_name = sanitize_key( mb_strtolower( (string) ( $entry['name'] ?? '' ), 'UTF-8' ) );
            if ( $stored_key === $entry_key || $stored_name === $entry_key ) return true;
        }
    }
    return false;
}

function uac_chat_verify_pin_or_claim_if_completed( $name, $pin ) {
    [ $valid_pin, $pin_or_msg ] = uac_validate_player_pin( $pin );
    if ( ! $valid_pin ) return [ false, $pin_or_msg, 'pin_required' ];

    [ $verify, $verify_msg, $verify_code ] = uac_player_pin_verify_only( $name, $pin_or_msg );
    if ( $verify === true ) return [ true, '', 'pin_ok' ];
    if ( $verify === false ) return [ false, $verify_msg, $verify_code ];

    
    if ( uac_chat_completed_by_km( $name ) || uac_chat_completed_by_password( $name ) || uac_chat_completed_by_daily( $name ) ) {
        return uac_player_pin_claim_or_verify( $name, $pin_or_msg, 'legacy_claim' );
    }

    return [ false, uac_chat_locked_message(), 'game_required' ];
}

function uac_chat_player_can_send( $name ) {
    if ( current_user_can( 'manage_options' ) ) return [ true, 'admin' ];

    $unlock = get_option( uac_chat_unlock_key( $name ), null );
    if ( is_array( $unlock ) && ! empty( $unlock['unlocked'] ) ) return [ true, sanitize_key( $unlock['last_game'] ?? 'game' ) ];

    if ( uac_chat_completed_by_km( $name ) ) {
        uac_chat_unlock_player( $name, 'klavye_muharebesi' );
        return [ true, 'klavye_muharebesi' ];
    }
    if ( uac_chat_completed_by_password( $name ) ) {
        uac_chat_unlock_player( $name, 'password_challenge' );
        return [ true, 'password_challenge' ];
    }
    if ( uac_chat_completed_by_daily( $name ) ) {
        uac_chat_unlock_player( $name, 'gunun_sirri' );
        return [ true, 'gunun_sirri' ];
    }

    return [ false, '' ];
}

function uac_chat_locked_message() {
    return 'Sohbete yazmak için önce aynı oyuncu adı ve PIN ile herhangi bir oyunu tamamlayıp skor/kayıt oluşturmalısın. Klavye Muharebesi, Password Challenge veya Günün Sırrı yeterli.';
}

function uac_chat_status( WP_REST_Request $request ) {
    $name = sanitize_text_field( $request->get_param( 'name' ) );
    [ $valid_name, $name_or_msg ] = uac_validate_name( $name, 20 );
    if ( ! $valid_name ) {
        return rest_ensure_response( [
            'ok'       => true,
            'can_chat' => false,
            'source'   => '',
            'message'  => 'Sohbet için önce geçerli bir oyuncu adı yazmalısın.',
        ] );
    }

    [ $pin_ok, $pin_msg, $pin_code ] = uac_chat_verify_pin_or_claim_if_completed( $name_or_msg, uac_pin_from_request( $request ) );
    if ( ! $pin_ok ) {
        return rest_ensure_response( [
            'ok'       => true,
            'can_chat' => false,
            'source'   => '',
            'code'     => $pin_code,
            'message'  => $pin_msg,
        ] );
    }

    [ $can_chat, $source ] = uac_chat_player_can_send( $name_or_msg );
    return rest_ensure_response( [
        'ok'       => true,
        'can_chat' => (bool) $can_chat,
        'source'   => $source,
        'message'  => $can_chat ? 'Sohbet yazma iznin açık.' : uac_chat_locked_message(),
    ] );
}

function uac_chat_send( WP_REST_Request $request ) {
    $ip = uac_get_real_ip();
    if ( uac_is_banned( $ip ) ) return uac_json_error( 'Erişiminiz engellenmiştir.', 'banned', 403 );

    
    $rate = uac_check_rate_limit( $ip, 'chat' );
    if ( $rate['blocked'] ) return uac_json_error( 'Çok hızlı mesaj gönderiyorsunuz.', 'rate_limited', 429, [ 'retry_after' => $rate['retry_after'] ] );

    
    $honeypot = sanitize_text_field( $request->get_param( 'website' ) );
    if ( $honeypot !== '' ) {
        uac_log_violation( $ip, 'chat_honeypot', $honeypot, 'honeypot' );
        return uac_json_error( 'Spam algılandı.', 'spam_detected', 403 );
    }

    $name = sanitize_text_field( $request->get_param( 'name' ) );
    $message = uac_chat_clean_message( $request->get_param( 'message' ) );
    $device_key = uac_chat_device_key( $request->get_param( 'device_key' ) );

    [ $valid_name, $name_or_msg ] = uac_validate_name( $name, 20 );
    if ( ! $valid_name ) return uac_json_error( $name_or_msg, 'invalid_name', 400 );
    $name = $name_or_msg;

    $banned_names = get_option( 'uac_banned_names', [] );
    if ( ! is_array( $banned_names ) ) $banned_names = [];
    $name_key = md5( uac_normalize_text( $name ) );
    $legacy_name_key = sanitize_key( mb_strtolower( $name, 'UTF-8' ) );
    if ( in_array( $name_key, $banned_names, true ) || in_array( $legacy_name_key, $banned_names, true ) ) {
        uac_log_violation( $ip, 'chat_banned_name', $name, 'banned_name' );
        return uac_json_error( 'Bu isim sohbetten engellenmiştir.', 'name_banned', 403 );
    }

    [ $pin_ok, $pin_msg, $pin_code ] = uac_chat_verify_pin_or_claim_if_completed( $name, uac_pin_from_request( $request ) );
    if ( ! $pin_ok ) return uac_json_error( $pin_msg, $pin_code, 403 );

    [ $can_chat, $chat_source ] = uac_chat_player_can_send( $name );
    if ( ! $can_chat ) {
        uac_log_violation( $ip, 'chat_locked', $name, 'game_required' );
        return uac_json_error( uac_chat_locked_message(), 'game_required', 403 );
    }

    if ( uac_strlen( $message ) < 1 || uac_strlen( $message ) > 120 ) return uac_json_error( 'Mesaj 1-120 karakter olmalıdır.', 'invalid_message', 400 );

    if ( preg_match( '~(https?://|www\.|\.com|\.net|\.org|\.xyz|\.io|\.gg|\.tr|discord\.gg|t\.me)~i', $message ) ) {
        uac_log_violation( $ip, 'chat_link', "$name | $message" );
        return uac_json_error( 'Sohbette link paylaşımı kapalı.', 'link_blocked', 400 );
    }

    [ $low_quality, $quality_msg ] = uac_chat_is_low_quality( $message );
    if ( $low_quality ) {
        uac_log_violation( $ip, 'chat_low_quality', "$name | $message", 'low_quality' );
        return uac_json_error( $quality_msg ?: 'Mesaj spam gibi görünüyor.', 'low_quality', 400 );
    }

    [ $bad_name_quality, $name_quality_msg ] = uac_chat_is_low_quality( $name );
    if ( $bad_name_quality && mb_strlen( $name, 'UTF-8' ) >= 4 ) {
        uac_log_violation( $ip, 'chat_low_quality_name', $name, 'low_quality_name' );
        return uac_json_error( 'Oyuncu adı rastgele harf dizisi gibi görünüyor.', 'low_quality_name', 400 );
    }

    $bad_name = uac_contains_bad_words( $name );
    $bad_msg  = uac_contains_bad_words( $message );
    if ( $bad_name || $bad_msg ) {
        uac_log_violation( $ip, 'bad_word', "$name | $message", $bad_name ?: $bad_msg );
        return uac_json_error( 'Uygunsuz kelime kullanımı tespit edildi.', 'bad_word', 400 );
    }

    $normalized_msg = trim( mb_strtolower( preg_replace( '/\s+/u', ' ', $message ), 'UTF-8' ) );
    $last_msg_key = 'uac_chat_last_msg_' . md5( $ip );
    $last_msg = get_transient( $last_msg_key );
    if ( $last_msg && hash_equals( (string) $last_msg, (string) $normalized_msg ) ) {
        return uac_json_error( 'Aynı mesajı üst üste gönderemezsiniz.', 'duplicate_message', 400 );
    }

    
    $cooldowns = [
        [ 'key' => 'uac_chat_ip_wait_' . md5( $ip ), 'seconds' => 3, 'message' => 'Çok hızlı gönderdin. Birkaç saniye bekle.' ],
        [ 'key' => 'uac_chat_name_wait_' . $name_key, 'seconds' => 4, 'message' => 'Bu isimle çok hızlı mesaj gönderildi. Birkaç saniye bekle.' ],
        [ 'key' => 'uac_chat_device_wait_' . md5( $device_key ), 'seconds' => 3, 'message' => 'Çok hızlı gönderdin. Birkaç saniye bekle.' ],
    ];
    foreach ( $cooldowns as $cd ) {
        $wait = uac_chat_recent_wait( $cd['key'], $cd['seconds'] );
        if ( $wait > 0 ) return uac_json_error( $cd['message'], 'chat_cooldown', 429, [ 'retry_after' => $wait ] );
    }

    $limits = [
        [ 'key' => 'uac_chat_ip_5m_' . md5( $ip ), 'limit' => 20, 'window' => 5 * MINUTE_IN_SECONDS ],
        [ 'key' => 'uac_chat_name_5m_' . $name_key, 'limit' => 14, 'window' => 5 * MINUTE_IN_SECONDS ],
        [ 'key' => 'uac_chat_device_5m_' . md5( $device_key ), 'limit' => 18, 'window' => 5 * MINUTE_IN_SECONDS ],
    ];
    foreach ( $limits as $lim ) {
        $retry = uac_chat_counter_check( $lim['key'], $lim['limit'], $lim['window'] );
        if ( $retry > 0 ) return uac_json_error( 'Kısa sürede çok fazla mesaj gönderildi. Biraz sonra tekrar dene.', 'chat_burst_limited', 429, [ 'retry_after' => $retry ] );
    }

    $post_id = wp_insert_post( [
        'post_type'   => 'pg_solution',
        'post_status' => 'publish',
        'post_title'  => 'chat_message',
    ] );
    if ( is_wp_error( $post_id ) || ! $post_id ) return uac_json_error( 'Mesaj kaydedilemedi.', 'save_failed', 500 );

    update_post_meta( $post_id, '_uac_kind', 'chat_message' );
    update_post_meta( $post_id, 'chat_name', $name );
    update_post_meta( $post_id, 'chat_message', $message );
    update_post_meta( $post_id, 'ip_hash', hash( 'sha256', $ip . wp_salt() ) );
    update_post_meta( $post_id, 'device_hash', hash( 'sha256', $device_key . wp_salt() ) );

    set_transient( $last_msg_key, $normalized_msg, 5 * MINUTE_IN_SECONDS );
    foreach ( $cooldowns as $cd ) uac_chat_touch_recent( $cd['key'], $cd['seconds'] );

    return rest_ensure_response( [ 'ok' => true ] );
}

function uac_chat_list( WP_REST_Request $request ) {
    $active_user = sanitize_text_field( $request->get_param( 'user' ) );
    $online_users = get_transient( 'uac_online_users' );
    if ( ! is_array( $online_users ) ) $online_users = [];
    $now = time();

    if ( $active_user !== '' && mb_strlen( $active_user, 'UTF-8' ) <= 20 ) $online_users[ $active_user ] = $now;
    foreach ( $online_users as $user => $ts ) if ( $now - (int) $ts > 15 ) unset( $online_users[ $user ] );
    set_transient( 'uac_online_users', $online_users, 45 );

    $posts = get_posts( [
        'post_type'   => 'pg_solution',
        'post_status' => 'publish',
        'numberposts' => 40,
        'orderby'     => 'date',
        'order'       => 'DESC',
        'meta_query'  => [ [ 'key' => '_uac_kind', 'value' => 'chat_message' ] ],
    ] );

    $messages = [];
    foreach ( array_reverse( $posts ) as $post ) {
        $chat_name = sanitize_text_field( get_post_meta( $post->ID, 'chat_name', true ) );
        $profile = get_option( uac_km_profile_key( $chat_name ), null );
        $best_score = 0;
        $rank_week = null;
        if ( is_array( $profile ) ) {
            $best_score = (int) ( $profile['best_score'] ?? 0 );
            $rank_week = uac_km_find_rank( 'uac_km_leaderboard_week_' . gmdate( 'oW' ), $chat_name );
        }

        $messages[] = [
            'id'            => (int) $post->ID,
            'name'          => $chat_name,
            'message'       => sanitize_text_field( get_post_meta( $post->ID, 'chat_message', true ) ),
            'time'          => get_date_from_gmt( get_gmt_from_date( $post->post_date ), 'H:i' ),
            'date'          => get_date_from_gmt( get_gmt_from_date( $post->post_date ), 'Y-m-d H:i' ),
            'km_best_score' => $best_score,
            'km_rank_week'  => $rank_week,
        ];
    }

    return rest_ensure_response( [
        'ok'           => true,
        'online_count' => max( 0, count( $online_users ) ),
        'messages'     => $messages,
    ] );
}

function uac_km_check_name( WP_REST_Request $request ) {

function uac_km_score_start( WP_REST_Request $request ) {

    $name = sanitize_text_field( $request->get_param( 'name' ) );
    if ( $name !== '' ) {
        [ $valid_name, $name_or_msg ] = uac_validate_name( $name, 20 );
        if ( ! $valid_name ) return uac_json_error( $name_or_msg, 'invalid_name', 400 );
        $name = $name_or_msg;

function uac_km_score_save( WP_REST_Request $request ) {
    $ip = uac_get_real_ip();
    if ( uac_is_banned( $ip ) ) return uac_json_error( 'Erişiminiz engellenmiştir.', 'banned', 403 );

    $rate = uac_check_rate_limit( $ip, 'score' );
    if ( $rate['blocked'] ) return uac_json_error( 'Çok fazla istek.', 'rate_limited', 429 );

    $name     = sanitize_text_field( $request->get_param( 'name' ) );
    $score    = max( 0, (int) $request->get_param( 'score' ) );
    $level    = max( 1, (int) $request->get_param( 'level' ) );
    $duration = max( 0, (int) $request->get_param( 'duration' ) );
    $words    = max( 0, (int) $request->get_param( 'words' ) );
    $session  = sanitize_text_field( $request->get_param( 'session' ) );

    [ $valid_name, $name_or_msg ] = uac_validate_name( $name, 20 );
    if ( ! $valid_name ) return uac_json_error( $name_or_msg, 'invalid_name', 400 );
    $name = $name_or_msg;

    $session_key = 'uac_km_session_' . $session;
    $session_data = get_transient( $session_key );
    if ( ! is_array( $session_data ) || ! hash_equals( (string) $session_data['ip'], (string) $ip ) || ! hash_equals( (string) $session_data['name'], (string) $name ) || ! empty( $session_data['used'] ) ) {
        uac_log_violation( $ip, 'cheat', "Geçersiz oyun oturumu: $name | $score", 'invalid_session' );
        return uac_json_error( 'Geçersiz oyun oturumu.', 'invalid_session', 400 );
    }
    $session_data['used'] = true;
    set_transient( $session_key, $session_data, 5 * MINUTE_IN_SECONDS );

    $server_elapsed = time() - (int) $session_data['start'];
    if ( $score <= 0 || $level < 1 || $words < 1 || $duration < 1 ) return uac_json_error( 'Geçersiz oyun verisi.', 'invalid_game_data', 400 );
    if ( $server_elapsed < 2 || $duration > $server_elapsed + 20 ) {
        uac_log_violation( $ip, 'cheat', "Süre uyuşmazlığı: {$duration}s / server {$server_elapsed}s", 'cheat_time' );
        return uac_json_error( 'Geçersiz oyun süresi.', 'cheat_detected', 400 );
    }

    $combo_score_margin = 3.0;     $max_by_level = $level * 6 * 70 * $combo_score_margin;
    $max_by_words = $words * 70 * $combo_score_margin;
    if ( $score > $max_by_level || $score > $max_by_words ) {
        uac_log_violation( $ip, 'cheat', "Şüpheli skor: $score | level: $level | words: $words", 'cheat_score' );
        return uac_json_error( 'Geçersiz skor.', 'cheat_detected', 400 );
    }
    if ( $words > 0 && $duration < max( 2, floor( $words * 0.45 ) ) ) {
        uac_log_violation( $ip, 'cheat', "Hız hilesi: $words kelime / {$duration}s", 'cheat_speed' );
        return uac_json_error( 'Geçersiz oyun verisi.', 'cheat_detected', 400 );
    }

    $now = current_time( 'mysql' );
    $profile_key = uac_km_profile_key( $name );
    $profile = get_option( $profile_key, [
        'name' => $name, 'total_games' => 0, 'total_score' => 0, 'best_score' => 0, 'best_level' => 0,
        'total_words' => 0, 'total_time' => 0, 'first_played' => $now, 'last_played' => $now,
    ] );

    $profile['total_games']++;
    $profile['total_score'] += $score;
    $profile['total_words'] += $words;
    $profile['total_time']  += $duration;
    $profile['last_played'] = $now;
    if ( $score > $profile['best_score'] ) $profile['best_score'] = $score;
    if ( $level > $profile['best_level'] ) $profile['best_level'] = $level;
    update_option( $profile_key, $profile, false );
    uac_chat_unlock_player( $name, 'klavye_muharebesi' );

    uac_km_save_to_board( 'uac_km_leaderboard_all', $name, $score, $level, $now, 100 );
    uac_km_save_to_board( 'uac_km_leaderboard_week_' . gmdate( 'oW' ), $name, $score, $level, $now, 100 );
    uac_km_save_to_board( 'uac_km_leaderboard_day_' . gmdate( 'Ymd' ), $name, $score, $level, $now, 100 );

    return rest_ensure_response( [ 'ok' => true, 'best_score' => $profile['best_score'], 'total_games' => $profile['total_games'] ] );
}

function uac_km_profile_key( $name ) {
    return 'uac_km_profile_' . sanitize_key( mb_strtolower( $name, 'UTF-8' ) );
}

function uac_km_save_to_board( $key, $name, $score, $level, $date, $limit = 100 ) {
    $board = get_option( $key, [] );
    if ( ! is_array( $board ) ) $board = [];
    $entry_key = sanitize_key( mb_strtolower( $name, 'UTF-8' ) );
    if ( ! isset( $board[ $entry_key ] ) || $score > (int) $board[ $entry_key ]['score'] ) {
        $board[ $entry_key ] = [ 'name' => $name, 'score' => $score, 'level' => $level, 'date' => $date ];
    }
    uasort( $board, fn( $a, $b ) => (int) $b['score'] <=> (int) $a['score'] );
    update_option( $key, array_slice( $board, 0, $limit, true ), false );
}

function uac_km_leaderboard( WP_REST_Request $request ) {
    $period = sanitize_key( $request->get_param( 'period' ) ?: 'all' );
    $limit = min( 50, max( 1, (int) ( $request->get_param( 'limit' ) ?: 10 ) ) );
    if ( $period === 'day' ) $key = 'uac_km_leaderboard_day_' . gmdate( 'Ymd' );
    elseif ( $period === 'week' ) $key = 'uac_km_leaderboard_week_' . gmdate( 'oW' );
    else $key = 'uac_km_leaderboard_all';

    $board = get_option( $key, [] );
    if ( ! is_array( $board ) ) $board = [];
    uasort( $board, fn( $a, $b ) => (int) $b['score'] <=> (int) $a['score'] );

    $entries = [];
    $rank = 1;
    foreach ( array_slice( $board, 0, $limit, true ) as $entry ) {
        $entries[] = [
            'rank'  => $rank++,
            'name'  => sanitize_text_field( $entry['name'] ?? '' ),
            'score' => (int) ( $entry['score'] ?? 0 ),
            'level' => (int) ( $entry['level'] ?? 1 ),
            'date'  => sanitize_text_field( $entry['date'] ?? '' ),
        ];
    }
    return rest_ensure_response( [ 'ok' => true, 'period' => $period, 'entries' => $entries ] );
}

function uac_km_get_profile( WP_REST_Request $request ) {

    if ( ! $valid_name ) return rest_ensure_response( [ 'ok' => false, 'found' => false ] );
    $name = $name_or_msg;

    $profile = get_option( uac_km_profile_key( $name ), null );
    if ( ! is_array( $profile ) ) return rest_ensure_response( [ 'ok' => true, 'found' => false ] );

    return rest_ensure_response( [
        'ok' => true, 'found' => true,
        'profile' => [
            'name' => sanitize_text_field( $profile['name'] ),
            'total_games' => (int) $profile['total_games'],

            'first_played' => sanitize_text_field( $profile['first_played'] ),
            'last_played' => sanitize_text_field( $profile['last_played'] ),
        ],
        'rank_all' => $rank_all,
        'rank_week' => $rank_week,
        'avg_score' => $avg_score,
    ] );
}

function uac_km_find_rank( $key, $name ) {
    $board = get_option( $key, [] );
    if ( ! is_array( $board ) ) return null;
    uasort( $board, fn( $a, $b ) => (int) $b['score'] <=> (int) $a['score'] );
    $needle = sanitize_key( mb_strtolower( $name, 'UTF-8' ) );
    $rank = 1;
    foreach ( $board as $key_name => $entry ) {
        if ( $key_name === $needle ) return $rank;
        $rank++;
    }
    return null;
}

        AND p.post_date < %s

            'name' => sanitize_text_field( get_post_meta( $post->ID, 'chat_name', true ) ),
            'message' => sanitize_text_field( get_post_meta( $post->ID, 'chat_message', true ) ),
            'date' => get_date_from_gmt( get_gmt_from_date( $post->post_date ), 'Y-m-d H:i' ),
            'ip_hash' => sanitize_text_field( get_post_meta( $post->ID, 'ip_hash', true ) ),
        ];
    }
    $banned_names = get_option( 'uac_banned_names', [] );
    if ( ! is_array( $banned_names ) ) $banned_names = [];
    return rest_ensure_response( [ 'ok' => true, 'items' => $items, 'banned_names' => array_values( $banned_names ) ] );
}

function uac_chat_admin_ban_name( WP_REST_Request $request ) {
    $name = sanitize_text_field( $request->get_param( 'name' ) );
    [ $valid, $name_or_msg ] = uac_validate_name( $name, 20 );
    if ( ! $valid ) return uac_json_error( $name_or_msg, 'invalid_name', 400 );
    $key = md5( uac_normalize_text( $name_or_msg ) );
    $legacy_key = sanitize_key( mb_strtolower( $name_or_msg, 'UTF-8' ) );
    $banned = get_option( 'uac_banned_names', [] );
    if ( ! is_array( $banned ) ) $banned = [];
    foreach ( array_filter( [ $key, $legacy_key ] ) as $ban_key ) {
        if ( ! in_array( $ban_key, $banned, true ) ) $banned[] = $ban_key;
    }
    update_option( 'uac_banned_names', array_values( $banned ), false );
    return rest_ensure_response( [ 'ok' => true, 'message' => 'İsim engellendi.', 'name_key' => $key ] );
}

function uac_chat_admin_unban_name( WP_REST_Request $request ) {
    $name = sanitize_text_field( $request->get_param( 'name' ) );
    $key = md5( uac_normalize_text( $name ) );
    $legacy_key = sanitize_key( mb_strtolower( $name, 'UTF-8' ) );
    $banned = get_option( 'uac_banned_names', [] );
    if ( ! is_array( $banned ) ) $banned = [];
    $banned = array_values( array_filter( $banned, fn( $item ) => $item !== $key && $item !== $legacy_key ) );
    update_option( 'uac_banned_names', $banned, false );
    return rest_ensure_response( [ 'ok' => true, 'message' => 'İsim engeli kaldırıldı.' ] );
}

function uac_daily_normalize( $text ) {
    $text = mb_strtolower( (string) $text, 'UTF-8' );
    $text = strtr( $text, [ 'ç' => 'c', 'ğ' => 'g', 'ı' => 'i', 'ö' => 'o', 'ş' => 's', 'ü' => 'u', 'â' => 'a', 'î' => 'i', 'û' => 'u' ] );
    $text = preg_replace( '/[^a-z0-9]+/u', '', $text );
    return $text;
}

function uac_daily_device_key( $device_key ) {
    $device_key = sanitize_text_field( (string) $device_key );
    if ( strlen( $device_key ) < 12 || strlen( $device_key ) > 120 ) return '';
    return $device_key;
}

function uac_daily_profile_key( $name, $device_key = '' ) {
    $name = mb_strtolower( trim( (string) $name ), 'UTF-8' );
    $seed = $name . '|' . ( $device_key ?: 'name-only' );
    return 'uac_daily_profile_' . hash( 'sha256', $seed . wp_salt( 'auth' ) );
}

function uac_daily_get_profile_by_name( $name ) {
    $board = get_option( 'uac_daily_rank_all', [] );
    if ( ! is_array( $board ) ) return null;
    $needle = sanitize_key( mb_strtolower( (string) $name, 'UTF-8' ) );
    foreach ( $board as $entry ) {
        if ( sanitize_key( mb_strtolower( $entry['name'] ?? '', 'UTF-8' ) ) === $needle && ! empty( $entry['profile_key'] ) ) {
            $profile = get_option( sanitize_text_field( $entry['profile_key'] ), null );
            if ( is_array( $profile ) ) return $profile;
        }
    }
    return null;
}

function uac_daily_assign_question( $profile, $profile_key ) {
    $today = uac_platform_now_date();
    if ( ( $profile['today_date'] ?? '' ) === $today && ! empty( $profile['today_id'] ) ) return $profile;

    $questions = uac_daily_questions();
    $used = is_array( $profile['used_ids'] ?? null ) ? $profile['used_ids'] : [];
    $candidates = array_values( array_filter( $questions, fn( $q ) => ! in_array( $q['id'], $used, true ) ) );
    if ( empty( $candidates ) ) {
        $profile['today_date'] = $today;
        $profile['today_id'] = '';
        $profile['today_done'] = true;
        $profile['season_completed'] = true;
        return $profile;
    }

    $idx = abs( crc32( $profile_key . '|' . $today ) ) % count( $candidates );
    $profile['today_date'] = $today;
    $profile['today_id'] = $candidates[ $idx ]['id'];
    $profile['today_done'] = false;
    $profile['today_attempts'] = 0;
    $profile['updated'] = current_time( 'mysql' );
    return $profile;
}

function uac_daily_question_by_id( $id ) {
    foreach ( uac_daily_questions() as $q ) if ( $q['id'] === $id ) return $q;
    return null;
}

function uac_daily_save_rank( $profile_key, $profile ) {
    $name = sanitize_text_field( $profile['name'] ?? '' );
    if ( $name === '' ) return;
    $rank_key = function_exists( 'uac_secret_v2_player_key' ) ? uac_secret_v2_player_key( $name ) : sanitize_key( uac_daily_normalize( $name ) );
    if ( $rank_key === '' ) $rank_key = sanitize_key( $profile_key );

    $entry = [
        'profile_key' => $profile_key,
        'name_key' => $rank_key,
        'name' => $name,
        'score' => (int) ( $profile['score'] ?? 0 ),
        'level' => (int) ( $profile['streak'] ?? 0 ),
        'answered' => (int) ( $profile['answered'] ?? 0 ),
        'date' => current_time( 'mysql' ),
    ];

    foreach ( [ 'uac_daily_rank_all', 'uac_daily_rank_week_' . uac_platform_week_key(), 'uac_daily_rank_day_' . uac_platform_day_key() ] as $key ) {
        $board = get_option( $key, [] );
        if ( ! is_array( $board ) ) $board = [];
        if ( function_exists( 'uac_secret_v2_compact_board' ) ) $board = uac_secret_v2_compact_board( $board );

        $old = $board[ $rank_key ] ?? null;
        if ( is_array( $old ) ) {
            $old_score = (int) ( $old['score'] ?? 0 );
            $old_answered = (int) ( $old['answered'] ?? 0 );
            if ( $old_score > $entry['score'] || ( $old_score === $entry['score'] && $old_answered > $entry['answered'] ) ) {
                $entry['score'] = $old_score;
                $entry['answered'] = $old_answered;
                $entry['level'] = max( (int) $entry['level'], (int) ( $old['level'] ?? 0 ) );
            }
        }

        $board[ $rank_key ] = $entry;
        if ( function_exists( 'uac_secret_v2_compact_board' ) ) {
            $board = uac_secret_v2_compact_board( $board );
        } else {
            uasort( $board, function( $a, $b ) {
                $score = (int) ( $b['score'] ?? 0 ) <=> (int) ( $a['score'] ?? 0 );
                if ( $score !== 0 ) return $score;
                return (int) ( $b['answered'] ?? 0 ) <=> (int) ( $a['answered'] ?? 0 );
            } );
        }
        update_option( $key, array_slice( $board, 0, 100, true ), false );
    }
}

function uac_daily_today( WP_REST_Request $request ) {
    $ip = uac_get_real_ip();
    if ( uac_is_banned( $ip ) ) return uac_json_error( 'Erişiminiz engellenmiştir.', 'banned', 403 );
    $rate = uac_check_rate_limit( $ip, 'daily' );
    if ( $rate['blocked'] ) return uac_json_error( 'Çok fazla istek.', 'rate_limited', 429 );

    [ $valid, $name_or_msg ] = uac_validate_name( $request->get_param( 'name' ), 20 );
    if ( ! $valid ) return uac_json_error( $name_or_msg, 'invalid_name', 400 );
    [ $pin_ok, $pin_msg, $pin_code ] = uac_request_claim_or_verify_pin( $request, $name_or_msg, 'gunun_sirri' );
    if ( ! $pin_ok ) return uac_json_error( $pin_msg, $pin_code, 403 );
    $device_key = uac_daily_device_key( $request->get_param( 'device_key' ) );
    $profile_key = uac_daily_profile_key( $name_or_msg, $device_key );
    $profile = get_option( $profile_key, uac_daily_default_profile( $name_or_msg ) );
    if ( ! is_array( $profile ) ) $profile = uac_daily_default_profile( $name_or_msg );
    $profile['name'] = $name_or_msg;
    $profile = uac_daily_assign_question( $profile, $profile_key );
    update_option( $profile_key, $profile, false );

    $question = ! empty( $profile['today_id'] ) ? uac_daily_question_by_id( $profile['today_id'] ) : null;
    return rest_ensure_response( [
        'ok' => true,
        'date' => uac_platform_now_date(),
        'season_completed' => ! empty( $profile['season_completed'] ),
        'question' => $question ? [
            'id' => $question['id'],
            'theme' => $question['theme'],
            'text' => $question['q'],
            'hint' => $question['hint'],
        ] : null,
        'profile' => uac_daily_public_profile_payload( $profile ),
        'leaderboard' => uac_platform_entries_from_board( 'uac_daily_rank_all', 10 ),
    ] );
}

function uac_daily_answer( WP_REST_Request $request ) {
    $ip = uac_get_real_ip();
    if ( uac_is_banned( $ip ) ) return uac_json_error( 'Erişiminiz engellenmiştir.', 'banned', 403 );
    $rate = uac_check_rate_limit( $ip, 'daily' );
    if ( $rate['blocked'] ) return uac_json_error( 'Çok fazla deneme.', 'rate_limited', 429 );

    [ $valid, $name_or_msg ] = uac_validate_name( $request->get_param( 'name' ), 20 );
    if ( ! $valid ) return uac_json_error( $name_or_msg, 'invalid_name', 400 );
    [ $pin_ok, $pin_msg, $pin_code ] = uac_request_claim_or_verify_pin( $request, $name_or_msg, 'gunun_sirri' );
    if ( ! $pin_ok ) return uac_json_error( $pin_msg, $pin_code, 403 );
    $device_key = uac_daily_device_key( $request->get_param( 'device_key' ) );
    $profile_key = uac_daily_profile_key( $name_or_msg, $device_key );
    $profile = get_option( $profile_key, uac_daily_default_profile( $name_or_msg ) );
    if ( ! is_array( $profile ) ) $profile = uac_daily_default_profile( $name_or_msg );
    $profile['name'] = $name_or_msg;
    $profile = uac_daily_assign_question( $profile, $profile_key );

    if ( ! empty( $profile['today_done'] ) ) {
        if ( (int) ( $profile['answered'] ?? 0 ) > 0 || (int) ( $profile['score'] ?? 0 ) > 0 ) uac_chat_unlock_player( $name_or_msg, 'gunun_sirri' );
        return rest_ensure_response( [ 'ok' => true, 'correct' => true, 'already_done' => true, 'message' => 'Bugünün sırrını zaten çözdün.', 'profile' => uac_daily_public_profile_payload( $profile ) ] );
    }
    $question = uac_daily_question_by_id( $profile['today_id'] ?? '' );
    if ( ! $question ) return uac_json_error( 'Bugün atanacak yeni soru kalmadı. Yeni soru havuzu eklenmeli.', 'season_completed', 409 );

    $answer = sanitize_text_field( $request->get_param( 'answer' ) );
    if ( uac_strlen( $answer ) < 1 || uac_strlen( $answer ) > 60 ) return uac_json_error( 'Cevap 1-60 karakter olmalıdır.', 'invalid_answer', 400 );
    if ( uac_contains_bad_words( $answer ) ) return uac_json_error( 'Uygunsuz cevap tespit edildi.', 'bad_word', 400 );

    $norm = uac_daily_normalize( $answer );
    $correct = false;
    foreach ( $question['a'] as $accepted ) {
        if ( $norm === uac_daily_normalize( $accepted ) ) { $correct = true; break; }
    }

    $profile['today_attempts'] = (int) ( $profile['today_attempts'] ?? 0 ) + 1;
    if ( $correct ) {
        $yesterday = gmdate( 'Y-m-d', time() - DAY_IN_SECONDS );
        $last = sanitize_text_field( $profile['last_correct_date'] ?? '' );
        $profile['streak'] = ( $last === $yesterday ) ? (int) ( $profile['streak'] ?? 0 ) + 1 : 1;
        $profile['best_streak'] = max( (int) ( $profile['best_streak'] ?? 0 ), (int) $profile['streak'] );
        $bonus = max( 0, 4 - (int) $profile['today_attempts'] );
        $profile['score'] = (int) ( $profile['score'] ?? 0 ) + 10 + ( (int) $profile['streak'] * 2 ) + $bonus;
        $profile['answered'] = (int) ( $profile['answered'] ?? 0 ) + 1;
        $profile['last_correct_date'] = uac_platform_now_date();
        $profile['today_done'] = true;
        $profile['used_ids'] = array_values( array_unique( array_merge( is_array( $profile['used_ids'] ?? null ) ? $profile['used_ids'] : [], [ $question['id'] ] ) ) );
        $profile['updated'] = current_time( 'mysql' );
        update_option( $profile_key, $profile, false );
        uac_daily_save_rank( $profile_key, $profile );
        uac_chat_unlock_player( $name_or_msg, 'gunun_sirri' );
        return rest_ensure_response( [
            'ok' => true,
            'correct' => true,
            'message' => 'Doğru cevap. Günlük seri devam ediyor.',
            'profile' => uac_daily_public_profile_payload( $profile ),
            'leaderboard' => uac_platform_entries_from_board( 'uac_daily_rank_all', 10 ),
        ] );
    }

    $profile['updated'] = current_time( 'mysql' );
    update_option( $profile_key, $profile, false );
    $tries_left = max( 0, 3 - (int) $profile['today_attempts'] );
    return rest_ensure_response( [
        'ok' => true,
        'correct' => false,
        'message' => $tries_left > 0 ? 'Yanlış cevap. Bir daha düşün.' : 'Yanlış. İpucunu iyi oku, denemeye devam edebilirsin.',
        'tries_left' => $tries_left,
        'hint' => $question['hint'],
        'profile' => uac_daily_public_profile_payload( $profile ),
    ] );
}

function uac_daily_profile( WP_REST_Request $request ) {
    [ $valid, $name_or_msg ] = uac_validate_name( $request->get_param( 'name' ), 20 );
    if ( ! $valid ) return uac_json_error( $name_or_msg, 'invalid_name', 400 );
    $device_key = uac_daily_device_key( $request->get_param( 'device_key' ) );
    $profile_key = uac_daily_profile_key( $name_or_msg, $device_key );
    $profile = get_option( $profile_key, null );
    if ( ! is_array( $profile ) ) return rest_ensure_response( [ 'ok' => true, 'found' => false ] );
    return rest_ensure_response( [ 'ok' => true, 'found' => true, 'profile' => uac_daily_public_profile_payload( $profile ), 'leaderboard' => uac_platform_entries_from_board( 'uac_daily_rank_all', 10 ) ] );
}

function uac_admin_chat_control() {
    if ( ! current_user_can( 'manage_options' ) ) return;
    if ( isset( $_POST['uac_delete_chat_id'] ) && check_admin_referer( 'uac_chat_control' ) ) {
        $id = absint( $_POST['uac_delete_chat_id'] );
        if ( $id && get_post_meta( $id, '_uac_kind', true ) === 'chat_message' ) {
            wp_delete_post( $id, true );
            echo '<div class="updated"><p>Mesaj silindi.</p></div>';
        }
    }
    if ( isset( $_POST['uac_ban_chat_name'] ) && check_admin_referer( 'uac_chat_control' ) ) {
        $name = sanitize_text_field( wp_unslash( $_POST['uac_ban_chat_name'] ) );
        $key = md5( uac_normalize_text( $name ) );
        $legacy_key = sanitize_key( mb_strtolower( $name, 'UTF-8' ) );
        $banned = get_option( 'uac_banned_names', [] );
        if ( ! is_array( $banned ) ) $banned = [];
        foreach ( array_filter( [ $key, $legacy_key ] ) as $ban_key ) {
            if ( ! in_array( $ban_key, $banned, true ) ) $banned[] = $ban_key;
        }
        update_option( 'uac_banned_names', array_values( $banned ), false );
        echo '<div class="updated"><p>İsim engellendi.</p></div>';
    }
    if ( isset( $_POST['uac_clear_chat'] ) && check_admin_referer( 'uac_chat_control' ) ) {
        $posts = get_posts( [ 'post_type' => 'pg_solution', 'post_status' => 'publish', 'numberposts' => 300, 'fields' => 'ids', 'meta_query' => [ [ 'key' => '_uac_kind', 'value' => 'chat_message' ] ] ] );
        foreach ( $posts as $id ) wp_delete_post( (int) $id, true );
        echo '<div class="updated"><p>Sohbet temizlendi.</p></div>';
    }
    $posts = get_posts( [ 'post_type' => 'pg_solution', 'post_status' => 'publish', 'numberposts' => 80, 'orderby' => 'date', 'order' => 'DESC', 'meta_query' => [ [ 'key' => '_uac_kind', 'value' => 'chat_message' ] ] ] );
    $banned = get_option( 'uac_banned_names', [] );
    if ( ! is_array( $banned ) ) $banned = [];
    echo '<div class="wrap"><h1>Sohbet Kontrol</h1><form method="post" style="margin:12px 0">';
    wp_nonce_field( 'uac_chat_control' );
    echo '<button class="button button-secondary" name="uac_clear_chat" value="1" onclick="return confirm(\'Sohbet temizlensin mi?\')">Sohbeti Temizle</button></form>';
    echo '<h2>Engelli İsimler</h2><p>' . ( empty( $banned ) ? 'Yok' : esc_html( implode( ', ', $banned ) ) ) . '</p>';
    echo '<table class="widefat striped"><thead><tr><th>Tarih</th><th>İsim</th><th>Mesaj</th><th>İşlem</th></tr></thead><tbody>';
    foreach ( $posts as $post ) {
        $name = sanitize_text_field( get_post_meta( $post->ID, 'chat_name', true ) );
        $msg = sanitize_text_field( get_post_meta( $post->ID, 'chat_message', true ) );
        echo '<tr><td>' . esc_html( $post->post_date ) . '</td><td>' . esc_html( $name ) . '</td><td>' . esc_html( $msg ) . '</td><td><form method="post" style="display:inline">';
        wp_nonce_field( 'uac_chat_control' );
        echo '<input type="hidden" name="uac_delete_chat_id" value="' . esc_attr( $post->ID ) . '"><button class="button">Sil</button></form> <form method="post" style="display:inline">';
        wp_nonce_field( 'uac_chat_control' );
        echo '<input type="hidden" name="uac_ban_chat_name" value="' . esc_attr( $name ) . '"><button class="button button-secondary">İsmi Engelle</button></form></td></tr>';
    }
    echo '</tbody></table></div>';
}

function uac_admin_daily() {
    if ( ! current_user_can( 'manage_options' ) ) return;
    $board = get_option( 'uac_daily_rank_all', [] );
    if ( ! is_array( $board ) ) $board = [];
    uasort( $board, fn( $a, $b ) => (int) ( $b['score'] ?? 0 ) <=> (int) ( $a['score'] ?? 0 ) );
    echo '<div class="wrap"><h1>Günün Sırrı</h1><p>Toplam soru havuzu: ' . esc_html( count( uac_daily_questions() ) ) . '. Aynı oyuncuya havuz bitene kadar aynı soru tekrar verilmez.</p>';
    echo '<table class="widefat striped"><thead><tr><th>#</th><th>Oyuncu</th><th>Puan</th><th>Doğru</th><th>Seri</th><th>Tarih</th></tr></thead><tbody>';
    $rank = 1;
    foreach ( array_slice( $board, 0, 100, true ) as $e ) {
        echo '<tr><td>' . esc_html( $rank++ ) . '</td><td>' . esc_html( $e['name'] ?? '' ) . '</td><td>' . esc_html( $e['score'] ?? 0 ) . '</td><td>' . esc_html( $e['answered'] ?? 0 ) . '</td><td>' . esc_html( $e['level'] ?? 0 ) . '</td><td>' . esc_html( $e['date'] ?? '' ) . '</td></tr>';
    }
    echo '</tbody></table></div>';
}

function uac_secret_v2_timezone() {
    
    
    try { return new DateTimeZone( 'Europe/Istanbul' ); }
    catch ( Exception $e ) { return wp_timezone(); }
}

function uac_secret_v2_day_key() {
    try {
        $dt = new DateTime( 'now', uac_secret_v2_timezone() );
        if ( (int) $dt->format( 'H' ) < 3 ) $dt->modify( '-1 day' );
        return $dt->format( 'Y-m-d' );
    } catch ( Exception $e ) {
        $ts = time() + 3 * HOUR_IN_SECONDS;
        if ( (int) gmdate( 'H', $ts ) < 3 ) $ts -= DAY_IN_SECONDS;
        return gmdate( 'Y-m-d', $ts );
    }
}

function uac_secret_v2_next_reset() {
    try {
        $dt = new DateTime( 'now', uac_secret_v2_timezone() );
        $reset = new DateTime( $dt->format( 'Y-m-d' ) . ' 03:00:00', uac_secret_v2_timezone() );
        if ( $dt >= $reset ) $reset->modify( '+1 day' );
        return $reset->format( DateTime::ATOM );
    } catch ( Exception $e ) {
        return gmdate( DateTime::ATOM, time() + DAY_IN_SECONDS );
    }
}

function uac_secret_v2_no_cache() {
    if ( ! headers_sent() ) {
        nocache_headers();
        header( 'Cache-Control: no-store, no-cache, must-revalidate, max-age=0' );
        header( 'Pragma: no-cache' );
        header( 'Expires: Wed, 11 Jan 1984 05:00:00 GMT' );
    }
}

function uac_secret_v2_custom_library() {
    $custom = get_option( 'uac_secret_v3_custom_packs', [] );
    return is_array( $custom ) ? $custom : [];
}

function uac_secret_v2_library() {
    return array_values( array_merge( uac_secret_v2_core_library(), uac_secret_v2_custom_library() ) );
}

function uac_secret_v2_flat_questions() {
    $flat = [];
    foreach ( uac_secret_v2_library() as $pack ) {
        foreach ( $pack['questions'] as $q ) {
            $q['pack_id'] = $pack['pack_id'];
            $q['pack_title'] = $pack['title'];
            $q['pack_theme'] = $pack['theme'];
            $flat[ $q['id'] ] = $q;
        }
    }
    return $flat;
}

function uac_secret_v2_library_stats() {
    $packs = uac_secret_v2_library();
    $total = 0; $concepts = []; $categories = [];
    foreach ( $packs as $p ) foreach ( $p['questions'] as $q ) {
        $total++;
        $concepts[ $q['concept_id'] ] = true;
        $categories[ $q['category'] ?? 'genel' ] = true;
    }
    return [ 'total_packs' => count( $packs ), 'total_questions' => $total, 'total_concepts' => count( $concepts ), 'total_categories' => count( $categories ) ];
}

function uac_secret_v2_device_key( $device_key ) {
    $device_key = sanitize_text_field( (string) $device_key );
    if ( strlen( $device_key ) < 8 || strlen( $device_key ) > 140 ) return 'name-only';
    return $device_key;
}

function uac_secret_v2_player_key( $name ) {
    $name = uac_secret_v2_normalize( (string) $name );
    return $name !== '' ? sanitize_key( $name ) : 'oyuncu';
}

function uac_secret_v2_profile_key( $name, $device_key = '' ) {
    
    
    return 'uac_secret_v2_profile_name_' . uac_secret_v2_player_key( $name );
}

function uac_secret_v2_legacy_profile_key( $name, $device_key = '' ) {
    
    $name = mb_strtolower( trim( (string) $name ), 'UTF-8' );
    $device_key = uac_secret_v2_device_key( $device_key );
    return 'uac_secret_v2_profile_' . hash( 'sha256', $name . '|' . $device_key . '|' . wp_salt( 'auth' ) );
}

function uac_secret_v2_profile_name_matches( $profile, $name ) {
    if ( ! is_array( $profile ) ) return false;
    $stored = sanitize_text_field( $profile['name'] ?? '' );
    if ( $stored === '' ) return false;
    return uac_secret_v2_player_key( $stored ) === uac_secret_v2_player_key( $name );
}

function uac_secret_v2_find_profiles_by_name( $name ) {
    global $wpdb;
    $wanted = uac_secret_v2_player_key( $name );
    $rows = $wpdb->get_results( "SELECT option_name, option_value FROM {$wpdb->options} WHERE option_name LIKE 'uac\_secret\_v2\_profile\_%' ESCAPE '\\'", ARRAY_A );
    $found = [];
    foreach ( $rows as $row ) {
        $profile = maybe_unserialize( $row['option_value'] ?? '' );
        if ( ! is_array( $profile ) ) continue;
        $stored_name = sanitize_text_field( $profile['name'] ?? '' );
        if ( $stored_name !== '' && uac_secret_v2_player_key( $stored_name ) === $wanted ) {
            $found[ $row['option_name'] ] = $profile;
        }
    }
    return $found;
}

function uac_secret_v2_normalize_history( $history ) {
    if ( ! is_array( $history ) ) return [];
    $unique = [];
    foreach ( $history as $row ) {
        if ( ! is_array( $row ) ) continue;
        $day = sanitize_text_field( $row['day_key'] ?? '' );
        $qid = sanitize_text_field( $row['question_id'] ?? '' );
        if ( $day === '' || $qid === '' ) continue;
        $key = $day . '|' . $qid;
        $score = max( 0, (int) ( $row['score'] ?? 0 ) );
        if ( ! isset( $unique[ $key ] ) || $score > (int) ( $unique[ $key ]['score'] ?? 0 ) ) {
            $row['day_key'] = $day;
            $row['question_id'] = $qid;
            $row['score'] = $score;
            $unique[ $key ] = $row;
        }
    }
    uasort( $unique, function( $a, $b ) {
        return strcmp( (string) ( $a['day_key'] ?? '' ), (string) ( $b['day_key'] ?? '' ) );
    } );
    return array_values( $unique );
}

function uac_secret_v2_recalculate_streak_from_history( $history ) {
    $days = [];
    foreach ( uac_secret_v2_normalize_history( $history ) as $row ) {
        $day = sanitize_text_field( $row['day_key'] ?? '' );
        if ( preg_match( '/^\d{4}-\d{2}-\d{2}$/', $day ) ) $days[ $day ] = true;
    }
    if ( empty( $days ) ) return [ 0, 0, '' ];
    $day_keys = array_keys( $days );
    sort( $day_keys );
    $best = 1;
    $cur = 1;
    for ( $i = 1; $i < count( $day_keys ); $i++ ) {
        $prev = strtotime( $day_keys[ $i - 1 ] . ' +1 day' );
        $now = strtotime( $day_keys[ $i ] );
        if ( gmdate( 'Y-m-d', $prev ) === $day_keys[ $i ] ) $cur++;
        else $cur = 1;
        $best = max( $best, $cur );
    }
    $last = end( $day_keys );
    $today = uac_secret_v2_day_key();
    $yesterday = gmdate( 'Y-m-d', strtotime( $today . ' -1 day' ) );
    $active = ( $last === $today || $last === $yesterday ) ? $cur : 0;
    return [ $active, $best, $last ];
}

function uac_secret_v2_merge_profiles( $base, $incoming, $name ) {
    if ( ! is_array( $base ) ) $base = uac_secret_v2_default_profile( $name );
    if ( ! is_array( $incoming ) ) return $base;

    $base['name'] = $name;

    foreach ( [ 'completed_packs', 'seen_questions', 'seen_concepts' ] as $field ) {
        $a = is_array( $base[ $field ] ?? null ) ? $base[ $field ] : [];
        $b = is_array( $incoming[ $field ] ?? null ) ? $incoming[ $field ] : [];
        $base[ $field ] = array_values( array_unique( array_merge( $a, $b ) ) );
    }

    $base_history = is_array( $base['history'] ?? null ) ? $base['history'] : [];
    $incoming_history = is_array( $incoming['history'] ?? null ) ? $incoming['history'] : [];
    $history = uac_secret_v2_normalize_history( array_merge( $base_history, $incoming_history ) );
    $base['history'] = array_slice( $history, -120 );
    $history_score = array_sum( array_map( fn( $row ) => max( 0, (int) ( $row['score'] ?? 0 ) ), $history ) );
    $history_answered = count( $history );

    $base['score'] = max( (int) ( $base['score'] ?? 0 ), (int) ( $incoming['score'] ?? 0 ), (int) $history_score );
    $base['answered'] = max( (int) ( $base['answered'] ?? 0 ), (int) ( $incoming['answered'] ?? 0 ), (int) $history_answered );

    [ $active_streak, $best_from_history, $last_day ] = uac_secret_v2_recalculate_streak_from_history( $history );
    $base['streak'] = max( (int) ( $base['streak'] ?? 0 ), (int) ( $incoming['streak'] ?? 0 ), (int) $active_streak );
    $base['best_streak'] = max( (int) ( $base['best_streak'] ?? 0 ), (int) ( $incoming['best_streak'] ?? 0 ), (int) $best_from_history );
    if ( $last_day !== '' ) $base['last_correct_day_key'] = $last_day;

    
    $incoming_updated = (string) ( $incoming['updated'] ?? '' );
    $base_updated = (string) ( $base['updated'] ?? '' );
    if ( ! empty( $incoming['today_day_key'] ) && ( empty( $base['today_day_key'] ) || strcmp( $incoming_updated, $base_updated ) >= 0 ) ) {
        foreach ( [ 'active_pack_id', 'current_day', 'today_day_key', 'today_question_id', 'today_done', 'today_attempts', 'today_hints', 'season_completed' ] as $field ) {
            if ( array_key_exists( $field, $incoming ) ) $base[ $field ] = $incoming[ $field ];
        }
    }

    
    if ( empty( $base['created'] ) || ( ! empty( $incoming['created'] ) && strcmp( (string) $incoming['created'], (string) $base['created'] ) < 0 ) ) {
        $base['created'] = sanitize_text_field( $incoming['created'] ?? current_time( 'mysql' ) );
    }

    $base['updated'] = current_time( 'mysql' );
    return $base;
}

function uac_secret_v2_pack_by_id( $pack_id ) {
    foreach ( uac_secret_v2_library() as $pack ) if ( $pack['pack_id'] === $pack_id ) return $pack;
    return null;

function uac_secret_v2_question_by_id( $id ) {
    $flat = uac_secret_v2_flat_questions();
    return $flat[ $id ] ?? null;
}

function uac_secret_v2_pick_new_pack( $profile ) {
    $completed = is_array( $profile['completed_packs'] ?? null ) ? $profile['completed_packs'] : [];
    foreach ( uac_secret_v2_library() as $pack ) {
        if ( ! in_array( $pack['pack_id'], $completed, true ) ) return $pack;
    }
    return null;
}

function uac_secret_v2_question_for_pack_day( $pack, $day_no, $profile ) {
    $seen_q = is_array( $profile['seen_questions'] ?? null ) ? $profile['seen_questions'] : [];
    $seen_c = is_array( $profile['seen_concepts'] ?? null ) ? $profile['seen_concepts'] : [];
    $same_day = array_values( array_filter( $pack['questions'], fn( $q ) => (int) $q['day_no'] === (int) $day_no ) );
    foreach ( $same_day as $q ) if ( ! in_array( $q['id'], $seen_q, true ) && ! in_array( $q['concept_id'], $seen_c, true ) ) return $q;
    foreach ( $same_day as $q ) if ( ! in_array( $q['id'], $seen_q, true ) ) return $q;
    return $same_day[0] ?? null;
}

function uac_secret_v2_assign_today( $profile ) {
    $today = uac_secret_v2_day_key();
    if ( ( $profile['today_day_key'] ?? '' ) === $today && ! empty( $profile['today_question_id'] ) ) return $profile;

    if ( empty( $profile['active_pack_id'] ) ) {
        $pack = uac_secret_v2_pick_new_pack( $profile );
        if ( ! $pack ) { $profile['season_completed'] = true; $profile['today_day_key'] = $today; $profile['today_question_id'] = ''; return $profile; }
        $profile['active_pack_id'] = $pack['pack_id'];
        $profile['current_day'] = 1;
    } else {
        $pack = uac_secret_v2_pack_by_id( $profile['active_pack_id'] );
        if ( ! $pack ) { $profile['active_pack_id'] = ''; return uac_secret_v2_assign_today( $profile ); }
        if ( ! empty( $profile['today_question_id'] ) ) $profile['current_day'] = min( 8, (int) ( $profile['current_day'] ?? 1 ) + 1 );
        if ( (int) $profile['current_day'] > 7 ) {
            $completed = is_array( $profile['completed_packs'] ?? null ) ? $profile['completed_packs'] : [];
            if ( ! in_array( $profile['active_pack_id'], $completed, true ) ) $completed[] = $profile['active_pack_id'];
            $profile['completed_packs'] = array_values( $completed );
            $pack = uac_secret_v2_pick_new_pack( $profile );
            if ( ! $pack ) { $profile['season_completed'] = true; $profile['today_day_key'] = $today; $profile['today_question_id'] = ''; return $profile; }
            $profile['active_pack_id'] = $pack['pack_id'];
            $profile['current_day'] = 1;
        }
    }

    $pack = uac_secret_v2_pack_by_id( $profile['active_pack_id'] );
    $question = $pack ? uac_secret_v2_question_for_pack_day( $pack, (int) $profile['current_day'], $profile ) : null;
    if ( ! $question ) { $profile['season_completed'] = true; return $profile; }

    $seen_q = is_array( $profile['seen_questions'] ?? null ) ? $profile['seen_questions'] : [];
    $seen_c = is_array( $profile['seen_concepts'] ?? null ) ? $profile['seen_concepts'] : [];
    if ( ! in_array( $question['id'], $seen_q, true ) ) $seen_q[] = $question['id'];
    if ( ! in_array( $question['concept_id'], $seen_c, true ) ) $seen_c[] = $question['concept_id'];
    $profile['seen_questions'] = array_values( $seen_q );
    $profile['seen_concepts'] = array_values( $seen_c );
    $profile['today_day_key'] = $today;
    $profile['today_question_id'] = $question['id'];
    $profile['today_done'] = false;
    $profile['today_attempts'] = 0;
    $profile['today_hints'] = 0;
    $profile['updated'] = current_time( 'mysql' );
    return $profile;
}

function uac_secret_v2_normalize( $text ) {
    $text = mb_strtolower( (string) $text, 'UTF-8' );
    $text = strtr( $text, [ 'ç'=>'c', 'ğ'=>'g', 'ı'=>'i', 'ö'=>'o', 'ş'=>'s', 'ü'=>'u', 'â'=>'a', 'î'=>'i', 'û'=>'u', 'é'=>'e' ] );
    $text = preg_replace( '/(.)\1{2,}/u', '$1$1', $text );
    $text = preg_replace( '/[^a-z0-9]+/u', '', $text );
    return trim( $text );
}

function uac_secret_v2_stems( $norm ) {
    $forms = [ $norm ];
    $suffixes = [ 'lar','ler','lari','leri','larda','lerde','lardan','lerden','im','in','um','un','dir','dur','tir','tur','lik','luk','ci','cu','de','da','den','dan','ye','ya' ];
    foreach ( $suffixes as $sfx ) {
        if ( strlen( $norm ) > strlen( $sfx ) + 3 && str_ends_with( $norm, $sfx ) ) $forms[] = substr( $norm, 0, -strlen( $sfx ) );
    }
    return array_values( array_unique( array_filter( $forms ) ) );
}

function uac_secret_v2_similarity( $a, $b ) {
    if ( $a === '' || $b === '' ) return 0;
    if ( $a === $b ) return 1;
    $max = max( strlen( $a ), strlen( $b ) );
    if ( $max < 4 ) return 0;
    $dist = levenshtein( $a, $b );
    return max( 0, 1 - ( $dist / $max ) );
}

function uac_secret_v2_match_answer( $answer, $exact, $close ) {
    $forms = uac_secret_v2_stems( uac_secret_v2_normalize( $answer ) );
    $exact_norm = array_map( 'uac_secret_v2_normalize', (array) $exact );
    $close_norm = array_map( 'uac_secret_v2_normalize', (array) $close );

    foreach ( $forms as $form ) {
        if ( in_array( $form, $exact_norm, true ) ) return [ 'result' => 'exact', 'score_base' => 100, 'matched' => $form ];
        if ( in_array( $form, $close_norm, true ) ) return [ 'result' => 'close', 'score_base' => 60, 'matched' => $form ];
    }
    foreach ( $forms as $form ) {
        if ( strlen( $form ) < 4 ) continue;
        foreach ( $exact_norm as $target ) if ( strlen( $target ) >= 4 && uac_secret_v2_similarity( $form, $target ) >= 0.87 ) return [ 'result' => 'exact', 'score_base' => 90, 'matched' => $target ];
        foreach ( $close_norm as $target ) if ( strlen( $target ) >= 4 && uac_secret_v2_similarity( $form, $target ) >= 0.78 ) return [ 'result' => 'close', 'score_base' => 55, 'matched' => $target ];
    }
    return [ 'result' => 'fail', 'score_base' => 0, 'matched' => '' ];
}

function uac_secret_v2_compact_board( $board ) {
    if ( ! is_array( $board ) ) return [];
    $compact = [];
    foreach ( $board as $row ) {
        $name = sanitize_text_field( $row['name'] ?? '' );
        if ( $name === '' ) continue;
        $key = uac_secret_v2_player_key( $name );
        $score = (int) ( $row['score'] ?? 0 );
        $answered = (int) ( $row['answered'] ?? 0 );
        $current = $compact[ $key ] ?? null;
        if ( ! $current || $score > (int) $current['score'] || ( $score === (int) $current['score'] && $answered > (int) ( $current['answered'] ?? 0 ) ) ) {
            $row['name'] = $name;
            $row['score'] = $score;
            $row['answered'] = $answered;
            $compact[ $key ] = $row;
        }
    }
    uasort( $compact, function( $a, $b ) {
        $score = (int) ( $b['score'] ?? 0 ) <=> (int) ( $a['score'] ?? 0 );
        if ( $score !== 0 ) return $score;
        return (int) ( $b['answered'] ?? 0 ) <=> (int) ( $a['answered'] ?? 0 );
    } );
    return $compact;
}

function uac_secret_v2_leaderboard( $limit = 10 ) {
    $board = uac_secret_v2_compact_board( get_option( 'uac_secret_v2_rank_all', [] ) );
    if ( count( $board ) !== count( get_option( 'uac_secret_v2_rank_all', [] ) ?: [] ) ) {
        update_option( 'uac_secret_v2_rank_all', array_slice( $board, 0, 150, true ), false );
    }
    $out = []; $rank = 1;
    foreach ( array_slice( $board, 0, $limit, true ) as $row ) {
        $out[] = [ 'rank' => $rank++, 'name' => sanitize_text_field( $row['name'] ?? '' ), 'score' => (int) ( $row['score'] ?? 0 ), 'answered' => (int) ( $row['answered'] ?? 0 ) ];
    }
    return $out;
}

function uac_secret_v2_save_rank( $profile_key, $profile ) {
    $name = sanitize_text_field( $profile['name'] ?? '' );
    if ( $name === '' ) return;
    $entry_key = uac_secret_v2_player_key( $name );
    $entry = [
        'profile_key' => $profile_key,
        'name_key' => $entry_key,
        'name' => $name,
        'score' => (int) ( $profile['score'] ?? 0 ),
        'answered' => (int) ( $profile['answered'] ?? 0 ),
        'level' => (int) ( $profile['streak'] ?? 0 ),
        'date' => current_time( 'mysql' ),
    ];
    foreach ( [ 'uac_secret_v2_rank_all', 'uac_daily_rank_all' ] as $key ) {
        $board = uac_secret_v2_compact_board( get_option( $key, [] ) );
        $old = $board[ $entry_key ] ?? null;
        if ( $old ) {
            $old_score = (int) ( $old['score'] ?? 0 );
            $old_answered = (int) ( $old['answered'] ?? 0 );
            if ( $old_score > $entry['score'] || ( $old_score === $entry['score'] && $old_answered > $entry['answered'] ) ) {
                $entry['score'] = $old_score;
                $entry['answered'] = $old_answered;
                $entry['level'] = max( (int) ( $entry['level'] ?? 0 ), (int) ( $old['level'] ?? 0 ) );
            }
        }
        $board[ $entry_key ] = $entry;
        $board = uac_secret_v2_compact_board( $board );
        update_option( $key, array_slice( $board, 0, 150, true ), false );
    }
}

function uac_secret_v2_load_profile_from_request( WP_REST_Request $request, &$profile_key = '', $require_pin = true ) {
    [ $valid, $name_or_msg ] = uac_validate_name( $request->get_param( 'name' ), 20 );
    if ( ! $valid ) return new WP_Error( 'invalid_name', $name_or_msg, [ 'status' => 400 ] );

    if ( $require_pin ) {
        [ $pin_ok, $pin_msg, $pin_code ] = uac_request_claim_or_verify_pin( $request, $name_or_msg, 'gunun_sirri' );
        if ( ! $pin_ok ) return new WP_Error( $pin_code, $pin_msg, [ 'status' => 403 ] );
    }

    $device = uac_secret_v2_device_key( $request->get_param( 'device_key' ) );
    $profile_key = uac_secret_v2_profile_key( $name_or_msg, $device );
    $profile = get_option( $profile_key, null );
    if ( ! is_array( $profile ) ) $profile = uac_secret_v2_default_profile( $name_or_msg );

    
    
    
    foreach ( uac_secret_v2_find_profiles_by_name( $name_or_msg ) as $key => $old_profile ) {
        if ( $key === $profile_key ) continue;
        $profile = uac_secret_v2_merge_profiles( $profile, $old_profile, $name_or_msg );
    }

    $legacy_key = uac_secret_v2_legacy_profile_key( $name_or_msg, $device );
    $legacy_profile = get_option( $legacy_key, null );
    if ( is_array( $legacy_profile ) ) $profile = uac_secret_v2_merge_profiles( $profile, $legacy_profile, $name_or_msg );

    $profile['name'] = $name_or_msg;
    $profile = uac_secret_v2_assign_today( $profile );
    update_option( $profile_key, $profile, false );
    uac_secret_v2_save_rank( $profile_key, $profile );
    return $profile;
}

function uac_secret_v2_today( WP_REST_Request $request ) {
    uac_secret_v2_no_cache();

    if ( uac_is_banned( $ip ) ) return uac_json_error( 'Erişiminiz engellenmiştir.', 'banned', 403 );
    $rate = uac_check_rate_limit( $ip, 'daily' );

    $profile_key = '';
    $profile = uac_secret_v2_load_profile_from_request( $request, $profile_key );
    if ( is_wp_error( $profile ) ) return uac_json_error( $profile->get_error_message(), $profile->get_error_code(), 400 );
    $q = ! empty( $profile['today_question_id'] ) ? uac_secret_v2_question_by_id( $profile['today_question_id'] ) : null;
    return rest_ensure_response( [
        'ok' => true,
        'day_key' => uac_secret_v2_day_key(),
        'next_reset' => uac_secret_v2_next_reset(),
        'season_completed' => ! empty( $profile['season_completed'] ),
        'question' => uac_secret_v2_public_question( $q ),
        'profile' => uac_secret_v2_public_profile( $profile ),
        'library' => uac_secret_v2_library_stats(),
        'leaderboard' => uac_secret_v2_leaderboard( 10 ),
    ] );
}

}

function uac_secret_v2_answer( WP_REST_Request $request ) {
    uac_secret_v2_no_cache();
    $ip = uac_get_real_ip();
    if ( uac_is_banned( $ip ) ) return uac_json_error( 'Erişiminiz engellenmiştir.', 'banned', 403 );
    $rate = uac_check_rate_limit( $ip, 'daily' );
    if ( $rate['blocked'] ) return uac_json_error( 'Çok fazla deneme.', 'rate_limited', 429 );
    $profile_key = '';
    $profile = uac_secret_v2_load_profile_from_request( $request, $profile_key );
    if ( is_wp_error( $profile ) ) return uac_json_error( $profile->get_error_message(), $profile->get_error_code(), 400 );
    if ( ! empty( $profile['today_done'] ) ) {
        if ( (int) ( $profile['answered'] ?? 0 ) > 0 || (int) ( $profile['score'] ?? 0 ) > 0 ) uac_chat_unlock_player( $profile['name'] ?? '', 'gunun_sirri' );
        return rest_ensure_response( [ 'ok'=>true, 'correct'=>true, 'already_done'=>true, 'message'=>'Bugünün sırrını zaten çözdün.', 'earned_score'=>0, 'profile'=>uac_secret_v2_public_profile( $profile ), 'library'=>uac_secret_v2_library_stats(), 'leaderboard'=>uac_secret_v2_leaderboard(10) ] );
    }
    $q = uac_secret_v2_question_by_id( $profile['today_question_id'] ?? '' );
    $posted_question_id = sanitize_text_field( $request->get_param( 'question_id' ) );
    if ( $posted_question_id !== '' ) {
        $posted_question = uac_secret_v2_question_by_id( $posted_question_id );
        if ( ! $posted_question ) return uac_json_error( 'Ekrandaki soru kütüphanede bulunamadı. Sayfayı yenile.', 'question_not_found', 409 );
        if ( ( $profile['today_day_key'] ?? '' ) !== uac_secret_v2_day_key() || ( $profile['today_question_id'] ?? '' ) !== $posted_question_id ) {
            $profile = uac_secret_v2_assign_today( $profile );
            update_option( $profile_key, $profile, false );
            return uac_json_error( 'Soru yenilenmiş. Lütfen Sırrı Getir butonuna basıp yeni soruyu aç.', 'question_expired', 409, [
                'day_key' => uac_secret_v2_day_key(),
                'profile' => uac_secret_v2_public_profile( $profile ),
                'question' => uac_secret_v2_public_question( uac_secret_v2_question_by_id( $profile['today_question_id'] ?? '' ) ),
                'library' => uac_secret_v2_library_stats(),
                'leaderboard' => uac_secret_v2_leaderboard(10),
            ] );
        }
        $q = $posted_question;
    }
    if ( ! $q ) return uac_json_error( 'Bugün atanacak yeni soru kalmadı. Yeni sır paketi eklenmeli.', 'season_completed', 409 );
    $answer = sanitize_text_field( $request->get_param( 'answer' ) );
    if ( uac_strlen( $answer ) < 1 || uac_strlen( $answer ) > 80 ) return uac_json_error( 'Cevap 1-80 karakter olmalıdır.', 'invalid_answer', 400 );
    if ( uac_contains_bad_words( $answer ) ) return uac_json_error( 'Uygunsuz cevap tespit edildi.', 'bad_word', 400 );
    $exact_answers = array_values( array_unique( array_merge( (array) ( $q['exact'] ?? [] ), [ $q['answer_key'] ?? '' ] ) ) );
    $match = uac_secret_v2_match_answer( $answer, $exact_answers, $q['close'] ?? [] );
    $profile['today_attempts'] = (int) ( $profile['today_attempts'] ?? 0 ) + 1;
    if ( $match['result'] !== 'fail' ) {
        $yesterday = gmdate( 'Y-m-d', strtotime( uac_secret_v2_day_key() . ' -1 day' ) );
        $last = sanitize_text_field( $profile['last_correct_day_key'] ?? '' );
        $profile['streak'] = ( $last === $yesterday ) ? (int) ( $profile['streak'] ?? 0 ) + 1 : 1;
        $profile['best_streak'] = max( (int) ( $profile['best_streak'] ?? 0 ), (int) $profile['streak'] );
        $hint_penalty = min( 50, (int) ( $profile['today_hints'] ?? 0 ) * 25 );
        $attempt_penalty = max( 0, ( (int) $profile['today_attempts'] - 1 ) * 10 );
        $final_bonus = ( (int) ( $q['day_no'] ?? 0 ) === 7 ) ? 200 : 0;
        $streak_bonus = min( 150, (int) $profile['streak'] * 5 );
        $earned = max( 20, (int) $match['score_base'] + $streak_bonus + $final_bonus - $hint_penalty - $attempt_penalty );
        $profile['score'] = (int) ( $profile['score'] ?? 0 ) + $earned;
        $profile['answered'] = (int) ( $profile['answered'] ?? 0 ) + 1;
        $profile['last_correct_day_key'] = uac_secret_v2_day_key();
        $profile['today_done'] = true;
        $history = is_array( $profile['history'] ?? null ) ? $profile['history'] : [];
        $history[] = [ 'day_key'=>uac_secret_v2_day_key(), 'question_id'=>$q['id'], 'pack_id'=>$q['pack_id'] ?? '', 'concept_id'=>$q['concept_id'] ?? '', 'result'=>$match['result'], 'score'=>$earned ];
        $profile['history'] = array_slice( $history, -80 );
        $profile['updated'] = current_time( 'mysql' );
        update_option( $profile_key, $profile, false );
        uac_secret_v2_save_rank( $profile_key, $profile );
        uac_chat_unlock_player( $profile['name'] ?? '', 'gunun_sirri' );
        return rest_ensure_response( [
            'ok'=>true, 'correct'=>true, 'result'=>$match['result'], 'message'=> $match['result']==='exact' ? 'Sır kabul edildi.' : 'Yakın cevap kabul edildi.', 'earned_score'=>$earned,
            'profile'=>uac_secret_v2_public_profile( $profile ), 'library'=>uac_secret_v2_library_stats(), 'leaderboard'=>uac_secret_v2_leaderboard(10)
        ] );
    }
    $profile['updated'] = current_time( 'mysql' );
    update_option( $profile_key, $profile, false );
    $hint_idx = min( 1, max( 0, (int) ( $profile['today_attempts'] ?? 1 ) - 1 ) );
    return rest_ensure_response( [
        'ok'=>true, 'correct'=>false, 'result'=>'fail', 'message'=>'Bu cevap kapıya değmedi. Tekrar düşün.', 'tries_left'=>max(0, 3 - (int)$profile['today_attempts']), 'hint'=>$q['hints'][$hint_idx] ?? $q['hints'][0] ?? '',
        'profile'=>uac_secret_v2_public_profile( $profile ), 'library'=>uac_secret_v2_library_stats()
    ] );
}

function uac_secret_v2_profile( WP_REST_Request $request ) {
    uac_secret_v2_no_cache();
    $profile_key = '';
    $profile = uac_secret_v2_load_profile_from_request( $request, $profile_key, false );
    if ( is_wp_error( $profile ) ) return rest_ensure_response( [ 'ok'=>true, 'found'=>false, 'message'=>$profile->get_error_message() ] );
    return rest_ensure_response( [ 'ok'=>true, 'found'=>true, 'profile'=>uac_secret_v2_public_profile( $profile ), 'library'=>uac_secret_v2_library_stats(), 'leaderboard'=>uac_secret_v2_leaderboard(10) ] );
}

function uac_secret_v2_repair_rank_boards() {
    $fixed = 0;
    $profiles_by_name = [];

    foreach ( uac_secret_v2_find_profiles_by_name( '' ) as $key => $noop ) {
        
    }

    global $wpdb;
    $rows = $wpdb->get_results( "SELECT option_name, option_value FROM {$wpdb->options} WHERE option_name LIKE 'uac\_secret\_v2\_profile\_%' ESCAPE '\\'", ARRAY_A );
    foreach ( $rows as $row ) {
        $profile = maybe_unserialize( $row['option_value'] ?? '' );
        if ( ! is_array( $profile ) ) continue;
        $name = sanitize_text_field( $profile['name'] ?? '' );
        if ( $name === '' ) continue;
        $pkey = uac_secret_v2_player_key( $name );
        if ( ! isset( $profiles_by_name[ $pkey ] ) ) $profiles_by_name[ $pkey ] = uac_secret_v2_default_profile( $name );
        $profiles_by_name[ $pkey ] = uac_secret_v2_merge_profiles( $profiles_by_name[ $pkey ], $profile, $name );
    }

    foreach ( $profiles_by_name as $pkey => $profile ) {
        $name = sanitize_text_field( $profile['name'] ?? '' );
        if ( $name === '' ) continue;
        $canonical_key = uac_secret_v2_profile_key( $name );
        update_option( $canonical_key, $profile, false );
        uac_secret_v2_save_rank( $canonical_key, $profile );
    }

    foreach ( [ 'uac_secret_v2_rank_all', 'uac_daily_rank_all' ] as $key ) {
        $before = get_option( $key, [] );
        $after = uac_secret_v2_compact_board( $before );
        update_option( $key, array_slice( $after, 0, 150, true ), false );
        $fixed += max( 0, count( is_array( $before ) ? $before : [] ) - count( $after ) );
    }
    return $fixed;
}

function uac_secret_v3_sanitize_import_packs( $data ) {
    if ( isset( $data['packs'] ) && is_array( $data['packs'] ) ) $data = $data['packs'];
    if ( isset( $data['questions'] ) && is_array( $data['questions'] ) ) {
        $chunks = array_chunk( $data['questions'], 7 );
        $packs = [];
        foreach ( $chunks as $i => $chunk ) {
            $packs[] = [ 'pack_id' => 'custom_' . ( $i + 1 ), 'title' => 'Özel Paket ' . ( $i + 1 ), 'theme' => 'İçe aktarılan sorular', 'questions' => $chunk ];
        }
        $data = $packs;
    }
    if ( ! is_array( $data ) ) return [];
    $packs = [];
    $pid = 1;
    foreach ( $data as $pack ) {
        if ( ! is_array( $pack ) ) continue;
        $questions = is_array( $pack['questions'] ?? null ) ? $pack['questions'] : [];
        if ( empty( $questions ) ) continue;
        $clean_questions = [];
        $day = 1;
        foreach ( array_slice( $questions, 0, 7 ) as $q ) {
            if ( ! is_array( $q ) ) continue;
            $question_text = sanitize_text_field( $q['q'] ?? $q['question'] ?? $q['question_text'] ?? '' );
            $answer = sanitize_text_field( $q['answer_key'] ?? $q['answer'] ?? '' );
            if ( $question_text === '' || $answer === '' ) continue;
            $aliases = array_values( array_filter( array_map( 'sanitize_text_field', (array) ( $q['aliases'] ?? $q['exact'] ?? [] ) ) ) );
            $close = array_values( array_filter( array_map( 'sanitize_text_field', (array) ( $q['close'] ?? [] ) ) ) );
            $hints = array_values( array_filter( array_map( 'sanitize_text_field', (array) ( $q['hints'] ?? [] ) ) ) );
            $concept = sanitize_key( $q['concept_id'] ?? uac_daily_normalize( $answer ) );
            $clean_questions[] = [
                'id' => sanitize_key( $q['id'] ?? ( 'custom_' . $pid . '_' . $day . '_' . md5( $question_text ) ) ),
                'day_no' => min( 7, max( 1, (int) ( $q['day_no'] ?? $day ) ) ),
                'concept_id' => strtoupper( $concept ),
                'answer_key' => $answer,
                'category' => sanitize_text_field( $q['category'] ?? 'özel' ),
                'difficulty' => sanitize_text_field( $q['difficulty'] ?? 'normal' ),
                'letter_count' => (int) ( $q['letter_count'] ?? mb_strlen( str_replace( ' ', '', $answer ), 'UTF-8' ) ),
                'symbol' => sanitize_text_field( $q['symbol'] ?? '◆' ),
                'theme' => sanitize_text_field( $q['theme'] ?? ( $q['category'] ?? 'özel' ) ),
                'q' => $question_text,
                'exact' => array_values( array_unique( array_merge( [ $answer ], $aliases ) ) ),
                'close' => $close,
                'hints' => $hints ?: [ 'Harf sayısına bak.', 'Kategori ipucunu kullan.' ],
                'lore' => sanitize_text_field( $q['lore'] ?? 'Özel kütüphaneden gelen soru.' ),
            ];
            $day++;
        }
        if ( $clean_questions ) {
            $packs[] = [
                'pack_id' => sanitize_key( $pack['pack_id'] ?? ( 'custom_pack_' . $pid ) ),
                'title' => sanitize_text_field( $pack['title'] ?? ( 'Özel Paket ' . $pid ) ),
                'theme' => sanitize_text_field( $pack['theme'] ?? 'İçe aktarılan sorular' ),
                'questions' => $clean_questions,
            ];
            $pid++;
        }
    }
    return $packs;
}

function uac_secret_v2_admin_page() {
    if ( ! current_user_can( 'manage_options' ) ) return;
    $notice = '';
    if ( isset( $_POST['uac_secret_v3_action'] ) && check_admin_referer( 'uac_secret_v3_import' ) ) {
        $action = sanitize_text_field( wp_unslash( $_POST['uac_secret_v3_action'] ) );
        if ( $action === 'clear_custom' ) {
            delete_option( 'uac_secret_v3_custom_packs' );
            $notice = 'Özel içe aktarılan paketler temizlendi. Çekirdek kütüphane duruyor.';
        } elseif ( $action === 'import_json' ) {
            $raw = (string) wp_unslash( $_POST['uac_secret_v3_json'] ?? '' );
            $decoded = json_decode( $raw, true );
            if ( ! is_array( $decoded ) ) {
                $notice = 'JSON okunamadı. Biçimi kontrol et.';
            } else {
                $packs = uac_secret_v3_sanitize_import_packs( $decoded );
                if ( empty( $packs ) ) $notice = 'Geçerli soru bulunamadı. question/q ve answer alanları zorunlu.';
                else {
                    update_option( 'uac_secret_v3_custom_packs', $packs, false );
                    $notice = count( $packs ) . ' özel paket içe aktarıldı.';
                }
            }
        }
    }
    $stats = uac_secret_v2_library_stats();
    $custom_count = count( uac_secret_v2_custom_library() );
    echo '<div class="wrap"><h1>Günün Sırrı v3 Kütüphane</h1>';
    if ( $notice ) echo '<div class="notice notice-info"><p>' . esc_html( $notice ) . '</p></div>';
    echo '<p>Bu sürüm geniş soru bankası kullanır: nesne tarifi, kelime anlamı, eş/zıt anlam, coğrafya, bilişim, matematik, bilmece, atasözü ve daha fazlası. Aynı oyuncuya aynı soru ID ve aynı konsept/cevap anahtarı tekrar verilmemeye çalışılır.</p>';
    echo '<p><b>Paket:</b> ' . esc_html( $stats['total_packs'] ) . ' · <b>Soru:</b> ' . esc_html( $stats['total_questions'] ) . ' · <b>Konsept/Cevap:</b> ' . esc_html( $stats['total_concepts'] ) . ' · <b>Kategori:</b> ' . esc_html( $stats['total_categories'] ?? 0 ) . ' · <b>Özel paket:</b> ' . esc_html( $custom_count ) . '</p>';
    echo '<hr><h2>JSON ile özel soru/paket içe aktar</h2>';
    echo '<p>JSON, çekirdek kütüphaneyi silmez; sadece özel paketleri değiştirir. 7 soruluk paketler önerilir. Düz questions listesi verirsen otomatik 7’li paketlere bölünür.</p>';
    echo '<form method="post">';
    wp_nonce_field( 'uac_secret_v3_import' );
    echo '<input type="hidden" name="uac_secret_v3_action" value="import_json">';
    echo '<textarea name="uac_secret_v3_json" rows="10" style="width:100%;font-family:monospace" placeholder="{&quot;questions&quot;:[{&quot;question&quot;:&quot;Bilgi hazinesi. 8 harf.&quot;,&quot;answer&quot;:&quot;dağarcık&quot;,&quot;category&quot;:&quot;kelime anlamı&quot;,&quot;hints&quot;:[&quot;8 harfli.&quot;,&quot;Kelime hazinesi anlamındadır.&quot;]}]}"></textarea>';
    echo '<p><button class="button button-primary">JSON İçe Aktar</button></p></form>';
    echo '<form method="post" onsubmit="return confirm(\'Özel paketler silinsin mi?\')">';
    wp_nonce_field( 'uac_secret_v3_import' );
    echo '<input type="hidden" name="uac_secret_v3_action" value="clear_custom"><button class="button">Özel Paketleri Temizle</button></form>';
    echo '<hr><h2>Kütüphane Önizleme</h2><table class="widefat striped"><thead><tr><th>Paket</th><th>Gün</th><th>Kategori</th><th>Harf</th><th>Cevap anahtarı</th><th>Soru</th></tr></thead><tbody>';
    $shown = 0;
    foreach ( uac_secret_v2_library() as $pack ) foreach ( $pack['questions'] as $q ) {
        if ( $shown++ >= 300 ) break 2;
        echo '<tr><td>' . esc_html( $pack['title'] ) . '</td><td>' . esc_html( $q['day_no'] ) . '</td><td>' . esc_html( $q['category'] ?? '' ) . '</td><td>' . esc_html( $q['letter_count'] ?? '' ) . '</td><td><b>' . esc_html( $q['answer_key'] ?? '' ) . '</b></td><td>' . esc_html( $q['q'] ) . '</td></tr>';
    }
    echo '</tbody></table></div>';
}
